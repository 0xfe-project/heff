module json (Json(..), stringify, repr, instanceShowJson, instanceToStringJson, Miss(..), ParseError, showParseError, instanceShowParseError, describe, instanceToStringParseError, Frame(..), Step(..), parse, field, at, str, int, bool, arr, obj, kv, num, jstr)

import rune as Rune

data Json = JNull | JBool Bool | JInt Int | JFloat Float64 | JString String | JArray (List Json) | JObject (List (String, Json))
stringify : Json -> String
repr : Json -> String
instance instanceShowJson : Show Json
instance instanceToStringJson : ToString Json
data Miss = Expected Int String
type ParseError = { offset : Int, line : Int, column : Int, expected : String }
showParseError : ParseError -> String
instance instanceShowParseError : Show ParseError
describe : ParseError -> String
instance instanceToStringParseError : ToString ParseError
data Frame = InArray (List Json) | InObject (List (String, Json)) String
data Step = Value Int (List Frame) | Got Json Int (List Frame) | Finished Json | Failed Miss
parse : String -> Either ParseError Json
field : String -> Json -> Maybe Json
at : List String -> Json -> Maybe Json
str : List String -> Json -> Maybe String
int : List String -> Json -> Maybe Int
bool : List String -> Json -> Maybe Bool
arr : List String -> Json -> Maybe (List Json)
obj : List (String, Json) -> Json
kv : String -> Json -> (String, Json)
num : Int -> Json
jstr : String -> Json
