module resource (With, scoped, held, inside, beside)

import co
import exception

data With s a
inside : {s a : Type} -> With s a -> a
beside : {s a b : Type} -> With s a -> b -> With s b
scoped : {e : HFunctor} -> {err r a : Type} -> {{ex : exception err <: e}} -> Co e (Either err r) -> (r -> Co e (Either err Unit)) -> ({s : Type} -> With s r -> Co e a) -> Co e a
held : {e : HFunctor} -> {r a : Type} -> Co e r -> (r -> Co e Unit) -> ({s : Type} -> With s r -> Co e a) -> Co e a
