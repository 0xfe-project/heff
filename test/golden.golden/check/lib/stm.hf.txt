module stm (stm, transaction, TVar, spawn, newTVar, atomically, readTVar, writeTVar, modifyTVar, retry, orElse, check, scheduled)

import co
import std.domain as Domain

data TVar (w : HFunctor) a
transaction : HFunctor -> HFunctor
readTVar : {w : HFunctor} -> {a : Type} -> TVar w a -> Co (transaction w) a
writeTVar : {w : HFunctor} -> {a : Type} -> TVar w a -> a -> Co (transaction w) Unit
modifyTVar : {w : HFunctor} -> {a : Type} -> TVar w a -> (a -> a) -> Co (transaction w) Unit
retry : {w : HFunctor} -> {a : Type} -> Co (transaction w) a
orElse : {w : HFunctor} -> {a : Type} -> Co (transaction w) a -> Co (transaction w) a -> Co (transaction w) a
check : {w : HFunctor} -> Bool -> Co (transaction w) Unit
stm : HFunctor -> HFunctor
spawn : {e : HFunctor} -> {w : HFunctor} -> {{sm : stm w <: e}} -> Co e Unit -> Co e Unit
newTVar : {e : HFunctor} -> {w : HFunctor} -> {a : Type} -> {{sm : stm w <: e}} -> a -> Co e (TVar w a)
atomically : {e : HFunctor} -> {w : HFunctor} -> {a : Type} -> {{sm : stm w <: e}} -> Co (transaction w) a -> Co e a
scheduled : {w : HFunctor} -> {a : Type} -> {{dm : Domain.domain <: w}} -> Co (stm w <+> w) a -> Co w (Maybe a)
