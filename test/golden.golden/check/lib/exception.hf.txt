module exception

import co

data ExceptionF err (m : Type -> Type) x = Throw err | Catch (m x) (err -> m x)
exception : (err : Type) -> HFunctor
throw : {err : Type} -> {e : HFunctor} -> {a : Type} -> {{ev : exception err <: e}} -> err -> Co e a
recover : {err : Type} -> {e : HFunctor} -> {a : Type} -> {{ev : exception err <: e}} -> Co e a -> (err -> Co e a) -> Co e a
data DivByZero = MkDivByZero
divOrThrow : {a : Type} -> {e : HFunctor} -> {{div : Division a}} -> {{z : IsZero a}} -> {{ev : exception DivByZero <: e}} -> a -> a -> Co e a
modOrThrow : {a : Type} -> {e : HFunctor} -> {{mod : Remainder a}} -> {{z : IsZero a}} -> {{ev : exception DivByZero <: e}} -> a -> a -> Co e a
triedFunctor : (err : Type) -> (e : HFunctor) -> RawFunctor
attempted : (err : Type) -> (e : HFunctor) -> Handler (exception err <+> e)
attempt : (err : Type) -> (e : HFunctor) -> {a : Type} -> Co (exception err <+> e) a -> Co e (Either err a)
