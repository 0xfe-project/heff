module ffi

data CType = CInteger String | CDouble | CBool | CString | CText | COwnedString | CBytes | COwnedBytes | CPointer String String | CVoid
data CResult = Gives CType | GivesOrErrno CType | DoneOrErrno
type Errno = { code : Int, name : String, message : String }
type Function = { name : String, args : List CType, result : CResult }
type Operation = { op : String, fun : Function }
data Described = Describe (List String) (List Operation)
type Links = { pkgConfig : List String = [], libs : List String = [], frameworks : List String = [] }
int : CType
int64 : CType
long : CType
size : CType
integer : String -> CType

text : CType
ownedString : CType
bytes : CType
ownedBytes : CType
pointer : String -> String -> CType
void : CType
gives : CType -> CResult
givesOrErrno : CType -> CResult
doneOrErrno : CResult
fun : String -> List CType -> CResult -> Function
operation : String -> Function -> Operation
describe : List String -> List Operation -> Described
