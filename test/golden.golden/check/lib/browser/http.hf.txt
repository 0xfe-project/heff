module browser.http

import co
import bytes as Bytes

type Request = { url : String, method : String = "GET", headers : List (Pair String String) = [], body : Bytes = Bytes.empty, timeout : Int = 0 }
type Response = { status : Int, headers : List (Pair String String), body : Bytes }
type HttpError = { message : String }
data HttpF (m : Type -> Type) x = Send Request (Either HttpError Response -> x)
http : HFunctor
send : {e : HFunctor} -> {{ev : http <: e}} -> Request -> Co e (Either HttpError Response)
get : {e : HFunctor} -> {{ev : http <: e}} -> String -> Co e (Either HttpError Response)
text : Response -> String
showError : HttpError -> String
