module (Document, Element, DomError, DomF(..), dom, withDocument, create, text, attr, append, clear, value, on, same, select, xpath, found, sleep, fetchText, textIn, showDomError)

import co
import with

data Document
data Element
type DomError = { message : String }
data DomF (m : Type -> Type) x = Open (Either DomError Document -> x) | Create Document String (Element -> x) | Text Element String x | Attr Element String String x | Append Element Element x | Clear Element x | Value Element (String -> x) | On Element String (String -> m Unit) x | Same Element Element (Bool -> x) | Select Document String (Maybe Element -> x) | XPath Document String (Maybe Element -> x) | Sleep Int x | Fetch String (String -> x)
dom : HFunctor
withDocument : {e : HFunctor} -> {a : Type} -> {{ev : dom <: e}} -> ({s : Type} -> With s Document -> Co e a) -> Co e (Either DomError a)
create : {e : HFunctor} -> {s : Type} -> {{ev : dom <: e}} -> With s Document -> String -> Co e Element
text : {e : HFunctor} -> {{ev : dom <: e}} -> Element -> String -> Co e Unit
attr : {e : HFunctor} -> {{ev : dom <: e}} -> Element -> String -> String -> Co e Unit
append : {e : HFunctor} -> {{ev : dom <: e}} -> Element -> Element -> Co e Unit
clear : {e : HFunctor} -> {{ev : dom <: e}} -> Element -> Co e Unit
value : {e : HFunctor} -> {{ev : dom <: e}} -> Element -> Co e String
on : {e : HFunctor} -> {{ev : dom <: e}} -> Element -> String -> (String -> Co e Unit) -> Co e Unit
same : {e : HFunctor} -> {{ev : dom <: e}} -> Element -> Element -> Co e Bool
select : {e : HFunctor} -> {s : Type} -> {{ev : dom <: e}} -> With s Document -> String -> Co e (Maybe Element)
xpath : {e : HFunctor} -> {s : Type} -> {{ev : dom <: e}} -> With s Document -> String -> Co e (Maybe Element)
found : {e : HFunctor} -> Maybe Element -> (Element -> Co e Unit) -> Co e Unit
sleep : {e : HFunctor} -> {{ev : dom <: e}} -> Int -> Co e Unit
fetchText : {e : HFunctor} -> {{ev : dom <: e}} -> String -> Co e String
textIn : {e : HFunctor} -> {s : Type} -> {{ev : dom <: e}} -> With s Document -> Element -> String -> String -> Co e Element
showDomError : DomError -> String
