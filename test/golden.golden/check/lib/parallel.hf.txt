runParallel : {a b x : Type} -> ParallelF a b (\y -> y) x -> x
turns : (a : Type) -> (b : Type) -> Handler (parallel a b)
bothSides : (Int -> Co (parallel Int Int <+> parallel Int Int) Int) -> Co (parallel Int Int <+> parallel Int Int) (Pair Int Int)
