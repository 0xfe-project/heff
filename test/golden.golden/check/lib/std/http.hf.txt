module std.http

import co
import bytes as Bytes

type Request = { url : String, method : String = "GET", headers : List (String, String) = [], body : Bytes = Bytes.empty, timeout : Int = 0, follow : Bool = True }
type Response = { status : Int, headers : List (String, String), body : Bytes }
type HttpError = { code : Int, message : String }
data HttpF (m : Type -> Type) x = Send Request (Either HttpError Response -> x)
http : HFunctor
send : {e : HFunctor} -> {{ev : http <: e}} -> Request -> Co e (Either HttpError Response)
get : {e : HFunctor} -> {{ev : http <: e}} -> String -> Co e (Either HttpError Response)
text : Response -> String
textOf : Either HttpError Response -> Either HttpError String
getText : {e : HFunctor} -> {{ev : http <: e}} -> String -> Co e (Either HttpError String)
lowerByte : Int -> Int
sameFrom : String -> String -> Int -> Bool
sameName : String -> String -> Bool
headerIn : String -> List (String, String) -> Maybe String
header : String -> Response -> Maybe String
showError : HttpError -> String
