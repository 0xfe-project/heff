module std.http

import co
import bytes as Bytes
import stream as Stream

type Request = { url : String, method : String = "GET", headers : List (String, String) = [], body : Bytes = Bytes.empty, timeout : Int = 0, connectTimeout : Int = 0, follow : Bool = True, idle : Int = 0 }
type Response = { status : Int, headers : List (String, String), body : Bytes }
type HttpError = { code : Int, message : String, url : String }
data Transfer
type Opened = { status : Int, headers : List (String, String), transfer : Transfer }
data HttpF (m : Type -> Type) x = Send Request (Either HttpError Response -> x) | Open Request (Either HttpError Opened -> x) | Piece Transfer (Either HttpError Bytes -> x)
http : HFunctor
type Streamed (e : HFunctor) = { status : Int, headers : List (String, String), body : Co (Stream.yielding Bytes <+> e) (Either HttpError Unit) }
piece : {e : HFunctor} -> {{ev : http <: e}} -> Transfer -> Co e (Either HttpError Bytes)
piecesOf : {e : HFunctor} -> {{ev : http <: e}} -> Transfer -> Co (Stream.yielding Bytes <+> e) (Either HttpError Unit)
streamedOf : {e : HFunctor} -> {{ev : http <: e}} -> Either HttpError Opened -> Either HttpError (Streamed e)
stream : {e : HFunctor} -> {{ev : http <: e}} -> Request -> Co e (Either HttpError (Streamed e))
gathered : {e : HFunctor} -> Co (Stream.yielding Bytes <+> e) (Either HttpError Unit) -> Co e (List Bytes, Either HttpError Unit)
wholeOf : {e : HFunctor} -> Streamed e -> (List Bytes, Either HttpError Unit) -> Either HttpError Response
readWhole : {e : HFunctor} -> Either HttpError (Streamed e) -> Co e (Either HttpError Response)
send : {e : HFunctor} -> {{ev : http <: e}} -> Request -> Co e (Either HttpError Response)
get : {e : HFunctor} -> {{ev : http <: e}} -> String -> Co e (Either HttpError Response)
text : Response -> String
textOf : String -> Either HttpError Response -> Either HttpError String
getText : {e : HFunctor} -> {{ev : http <: e}} -> String -> Co e (Either HttpError String)
lowerByte : Int -> Int
sameName : String -> String -> Bool
headerIn : String -> List (String, String) -> Maybe String
header : String -> Response -> Maybe String
instance instanceShowHttpError : Show HttpError
