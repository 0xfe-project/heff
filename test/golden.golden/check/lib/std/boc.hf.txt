module std.boc (Cown, Cowns, instanceCownsPosition, instanceCownsPair, Behaviours, boc, when, whenever, drained, opening, steps, runBoc)

import co
import green
import state
import std.store
import std.ordmap as OrdMap

type Cown s a = Position s a
type Behaviours s = { now : s, queues : Queues, live : Int, idle : List Int }
boc : Type -> HFunctor
opening : {s : Type} -> s -> Behaviours s
steps : {s : Type} -> Behaviours s -> Int

theory Cowns c where
  Store : Type
  Contents : Type
  position : c -> Position Store Contents

instance instanceCownsPosition : {s a : Type} -> Cowns (Position s a)
instance instanceCownsPair : {c d : Type} -> {{l : Cowns c}} -> {{r : Cowns d @ { Store = l.Store }}} -> Cowns (c, d)
when : {c : Type} -> {e : HFunctor} -> c -> {{cw : Cowns c}} -> {{gr : green <: e}} -> {{bc : boc cw.Store <: e}} -> (cw.Contents -> Co e cw.Contents) -> Co e Unit
whenever : {c : Type} -> {e : HFunctor} -> c -> {{cw : Cowns c}} -> {{gr : green <: e}} -> {{bc : boc cw.Store <: e}} -> (cw.Contents -> Bool) -> (cw.Contents -> Co e cw.Contents) -> Co e Unit
drained : {s : Type} -> {e : HFunctor} -> {{gr : green <: e}} -> {{bc : boc s <: e}} -> Co e Unit
runBoc : {s a : Type} -> s -> ({e : HFunctor} -> {{gr : green <: e}} -> {{bc : boc s <: e}} -> Co e a) -> Maybe (a, s)
