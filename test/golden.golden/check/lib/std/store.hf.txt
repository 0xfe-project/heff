module std.store (Position, Part, whole, fstOf, sndOf, left, right, first, second, third, both, contents, replaced, parts, overlaps, enclosing, instanceEqualityPart, instanceOrderPart, greenWith)

import co
import green
import state

data Part
data Position s a
whole : {s : Type} -> Position s s
fstOf : {s a b : Type} -> Position s (a, b) -> Position s a
sndOf : {s a b : Type} -> Position s (a, b) -> Position s b
left : {a b : Type} -> Position (a, b) a
right : {a b : Type} -> Position (a, b) b
first : {a b c : Type} -> Position (a, b, c) a
second : {a b c : Type} -> Position (a, b, c) b
third : {a b c : Type} -> Position (a, b, c) c
both : {s a b : Type} -> Position s a -> Position s b -> Position s (a, b)
contents : {s a : Type} -> Position s a -> s -> a
replaced : {s a : Type} -> Position s a -> a -> s -> s
parts : {s a : Type} -> Position s a -> List Part
overlaps : Part -> Part -> Bool
enclosing : Part -> List Part
instance instanceEqualityPart : Equality Part
instance instanceOrderPart : Order Part
greenWith : {x a : Type} -> x -> ({e : HFunctor} -> {{gr : green <: e}} -> {{st : state x <: e}} -> Co e a) -> (Maybe a, x)
