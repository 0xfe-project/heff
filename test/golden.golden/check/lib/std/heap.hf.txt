module std.heap (Heap, empty, singleton, null, size, merge, insert, findMin, deleteMin, popMin, fromList, toList)

data Heap a
empty : {a : Type} -> Heap a
singleton : {a : Type} -> a -> Heap a
null : {a : Type} -> Heap a -> Bool
size : {a : Type} -> Heap a -> Int
merge : {a : Type} -> {{ord : Order a}} -> Heap a -> Heap a -> Heap a
insert : {a : Type} -> {{ord : Order a}} -> Heap a -> a -> Heap a
findMin : {a : Type} -> Heap a -> Maybe a
deleteMin : {a : Type} -> {{ord : Order a}} -> Heap a -> Heap a
popMin : {a : Type} -> {{ord : Order a}} -> Heap a -> Maybe (a, Heap a)
fromList : {a : Type} -> {{ord : Order a}} -> List a -> Heap a
toList : {a : Type} -> {{ord : Order a}} -> Heap a -> List a
