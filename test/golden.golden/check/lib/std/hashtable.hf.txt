import array as Array
import int as Int
import std.hashmap as HashMap

lookup : {k v : Type} -> {{hashing : HashMap.Hash k}} -> {{equality : Equality k}} -> HashTable k v -> k -> Maybe v
remove : {k v : Type} -> {{hashing : HashMap.Hash k}} -> {{equality : Equality k}} -> HashTable k v -> k -> (Maybe v, HashTable k v)
delete : {k v : Type} -> {{hashing : HashMap.Hash k}} -> {{equality : Equality k}} -> HashTable k v -> k -> HashTable k v
alter : {k v : Type} -> {{hashing : HashMap.Hash k}} -> {{equality : Equality k}} -> HashTable k v -> k -> (Maybe v -> Maybe v) -> HashTable k v
foldl : {k v b : Type} -> (b -> k -> v -> b) -> b -> HashTable k v -> b
size : {k v : Type} -> HashTable k v -> Int
toList : {k v : Type} -> HashTable k v -> List (k, v)
member : {k v : Type} -> {{hashing : HashMap.Hash k}} -> {{equality : Equality k}} -> HashTable k v -> k -> Bool
keys : {k v : Type} -> HashTable k v -> List k
values : {k v : Type} -> HashTable k v -> List v

instance instanceShowHashTable : {k v : Type} -> {{showsKey : Show k}} -> {{showsValue : Show v}} -> Show (HashTable k v)
updateBy : {k v q : Type} -> {{view : HashMap.KeyView k q}} -> HashTable k v -> q -> v -> (v -> v -> v) -> HashTable k v
lookupBy : {k v q : Type} -> {{view : HashMap.KeyView k q}} -> HashTable k v -> q -> Maybe v
