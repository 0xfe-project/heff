module std.terminal.widget (Key(..), Decoded, decode, columns, width, cut, pad, wrap, Style(..), Span, Row, Frame, Size, plain, bold, dim, inverse, rowText, texts, fill, status, paint, Menu, menu, moveBy, menuKey, menuRows, Field, field, fieldText, fieldCursor, fieldKey, Choice, choice, chosen, choiceKey, choiceRow, Step(..), App, stepAll, advance, replay, shownAfter)

import rune as Rune

data Key = Char Rune | Enter | Escape | Tab | BackTab | Backspace | Delete | ArrowUp | ArrowDown | ArrowLeft | ArrowRight | Home | End | PageUp | PageDown | Control Rune | Sequence String
type Decoded = { key : Key, next : Int }
decode : String -> List Key
columns : Rune -> Int
width : String -> Int
cut : Int -> String -> String
pad : Int -> String -> String
wrap : Int -> String -> List String
data Style = Plain | Bold | Dim | Inverse | Underline
type Span = { text : String, style : Style }
type Row = List Span
type Frame = { rows : List Row, cursor : Maybe (Int, Int) }
type Size = { columns : Int, rows : Int }
plain : String -> Span
bold : String -> Span
dim : String -> Span
inverse : String -> Span
rowText : Row -> String
texts : Frame -> List String
fill : Int -> List Row -> List Row
status : Int -> String -> Row
paint : Size -> Frame -> String
type Menu = { at : Int }
menu : Menu
moveBy : Int -> Int -> Menu -> Menu
menuKey : Int -> Key -> Menu -> Menu
menuRows : Int -> Menu -> List Row -> List Row
type Field = { text : List Rune, at : Int }
field : String -> Field
fieldText : Field -> String
fieldCursor : Field -> Int
fieldKey : Key -> Field -> Field
type Choice = { options : List String, at : Int }
choice : List String -> Choice
chosen : Choice -> String
choiceKey : Key -> Choice -> Choice
choiceRow : Bool -> (String -> String) -> Choice -> Row
data Step s r = Continue s | Done r
type App s r = { start : s, step : Key -> s -> Step s r, view : Size -> s -> Frame }
stepAll : {s r : Type} -> (Key -> s -> Step s r) -> Step s r -> List Key -> Step s r
advance : {s r : Type} -> (Key -> s -> Step s r) -> s -> Maybe String -> Step s (Maybe r)
replay : {s r : Type} -> App s r -> List Key -> Either s r
shownAfter : {s r : Type} -> App s r -> Size -> List Key -> List String
