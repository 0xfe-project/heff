module (withFile, readLineOf, writeStringTo)

import co
import std.filesystem
import with

withFile : {e : HFunctor} -> {a : Type} -> {{ev : filesystem <: e}} -> String -> OpenOptions -> ({s : Type} -> With s OpenFile -> Co e a) -> Co e (Either FileError a)
readLineOf : {e : HFunctor} -> {s : Type} -> {{ev : filesystem <: e}} -> With s OpenFile -> Co e (Either FileError (Maybe String))
writeStringTo : {e : HFunctor} -> {s : Type} -> {{ev : filesystem <: e}} -> With s OpenFile -> String -> Co e (Either FileError Unit)
