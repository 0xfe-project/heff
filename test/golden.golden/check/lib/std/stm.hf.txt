module std.stm

import co
import green
import state

data TVar s a = MkTVar (s -> a) (a -> s -> s)
whole : {s : Type} -> TVar s s
fstOf : {s a b : Type} -> TVar s (a, b) -> TVar s a
sndOf : {s a b : Type} -> TVar s (a, b) -> TVar s b
left : {a b : Type} -> TVar (a, b) a
right : {a b : Type} -> TVar (a, b) b
first : {a b c : Type} -> TVar (a, b, c) a
second : {a b c : Type} -> TVar (a, b, c) b
third : {a b c : Type} -> TVar (a, b, c) c
data StmF s (m : Type -> Type) x = Look (s -> x) | Change (s -> s) x | Retry | OrElse (m x) (m x)
stm : Type -> HFunctor
readTVar : {s a : Type} -> TVar s a -> Co (stm s) a
writeTVar : {s a : Type} -> TVar s a -> a -> Co (stm s) Unit
modifyTVar : {s a : Type} -> TVar s a -> (a -> a) -> Co (stm s) Unit
retry : {s a : Type} -> Co (stm s) a
orElse : {s a : Type} -> Co (stm s) a -> Co (stm s) a -> Co (stm s) a
check : {s : Type} -> Bool -> Co (stm s) Unit
data Ran s x = Committed x s Bool | Retried
thenRan : {s x y : Type} -> Ran s x -> (x -> s -> Bool -> Ran s y) -> Ran s y
orElseRan : {s x : Type} -> Ran s x -> (s -> Bool -> Ran s x) -> s -> Bool -> Ran s x
storeAlg : {s x : Type} -> StmF s (\y -> s -> Bool -> Ran s y) x -> s -> Bool -> Ran s x
onStore : (s : Type) -> Handler (stm s)
transacted : {s a : Type} -> s -> Co (stm s) a -> Ran s a
type Shared s = { now : s, waiting : List Int }
stored : Type -> HFunctor
sharing : {s : Type} -> s -> Shared s
nowOf : {s : Type} -> Shared s -> s
wakeAll : {e : HFunctor} -> {{gr : green <: e}} -> List Int -> Co e Unit
settled : {s : Type} -> {e : HFunctor} -> {{gr : green <: e}} -> {{st : state (Shared s) <: e}} -> {a : Type} -> (Maybe Int -> Co e a) -> Shared s -> Maybe Int -> Ran s a -> Co e a
atomicallyHolding : {s : Type} -> {e : HFunctor} -> {{gr : green <: e}} -> {{st : state (Shared s) <: e}} -> {a : Type} -> Maybe Int -> Co (stm s) a -> Co e a
atomically : {s : Type} -> {e : HFunctor} -> {{gr : green <: e}} -> {{st : state (Shared s) <: e}} -> {a : Type} -> Co (stm s) a -> Co e a
data NoneF (m : Type -> Type) x = NoneOf (NoneF m x)
none : HFunctor
noneAlg : {x : Type} -> NoneF (\y -> y) x -> x
ranAlone : Handler none
runGreen : {s a : Type} -> s -> ({e : HFunctor} -> {{gr : green <: e}} -> {{st : stored s <: e}} -> Co e a) -> Maybe a
