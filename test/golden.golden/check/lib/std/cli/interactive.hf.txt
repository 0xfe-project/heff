module std.cli.interactive (Entry, entries, Part(..), Input(..), Slot, Form, formOf, formWords, formProblem, lacking, quoted, commandLine, Listing, pathsFor, Model, start, matching, picked, Result, step, view, app, prompt, listed)

import co
import exception
import std.filesystem
import rune as Rune
import std.cli as CLI
import std.path as Path
import std.terminal.widget as Widget

type Entry = { name : String, doc : String, shape : CLI.Shape }
entries : {r : Type} -> List (CLI.Command r) -> List Entry
data Part = Positional CLI.Pos | Flagged CLI.Flag
data Input = Typed Widget.Field | Picked Widget.Choice
type Slot = { part : Part, input : Input }
type Form = { entry : Entry, slots : List Slot, at : Int }
formOf : Entry -> Form
formWords : Form -> List String
formProblem : Form -> Maybe CLI.Problem
lacking : List Entry -> List String -> Maybe Form
quoted : String -> String
commandLine : String -> String -> List String -> String
type Listing = { dir : String, names : List String }
pathsFor : CLI.Complete -> Listing -> String -> List String
type Model = { prog : String, entries : List Entry, filter : Widget.Field, typing : Bool, menu : Widget.Menu, form : Maybe Form, alone : Bool, listing : Maybe Listing }
start : String -> List Entry -> Model
matching : Model -> List Entry
picked : Model -> Maybe Entry
type Result = Maybe (List String)
step : Widget.Key -> Model -> Widget.Step Model Result
view : Widget.Size -> Model -> Widget.Frame
app : String -> List Entry -> Widget.App Model Result
prompt : String -> List Entry -> Form -> Widget.App Model Result
listed : {e : HFunctor} -> {{disk : filesystem <: e}} -> Model -> Co e Model
