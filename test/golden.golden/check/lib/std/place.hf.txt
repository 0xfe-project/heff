import co

data Spot
data Place a = MkPlace Spot
data Channel
data PlaceChannel a = MkPlaceChannel Channel
type Started (m : Type -> Type) x = (r : Type) -> ((a : Type) -> m a -> (Place a -> x) -> r) -> r
type Got x = (r : Type) -> ((a : Type) -> (a -> x) -> r) -> r
type Sent x = (r : Type) -> ((a : Type) -> a -> x -> r) -> r
data PlacesF (m : Type -> Type) x = Spawn (Started m x) | Join Spot (Got x) | NewChannel (Channel -> x) | Send Channel (Sent x) | Recv Channel (Got x)
started : {m : Type -> Type} -> {a x : Type} -> m a -> (Place a -> x) -> Started m x
got : {a x : Type} -> (a -> x) -> Got x
sent : {a x : Type} -> a -> x -> Sent x
gotThen : {x y : Type} -> (x -> y) -> Got x -> Got y
places : HFunctor
spawnPlace : {e : HFunctor} -> {{pl : places <: e}} -> {a : Type} -> Co e a -> Co e (Place a)
joinPlace : {e : HFunctor} -> {{pl : places <: e}} -> {a : Type} -> Place a -> Co e a
newPlaceChannel : {e : HFunctor} -> {{pl : places <: e}} -> {a : Type} -> Co e (PlaceChannel a)
sendPlace : {e : HFunctor} -> {{pl : places <: e}} -> {a : Type} -> PlaceChannel a -> a -> Co e Unit
recvPlace : {e : HFunctor} -> {{pl : places <: e}} -> {a : Type} -> PlaceChannel a -> Co e a
