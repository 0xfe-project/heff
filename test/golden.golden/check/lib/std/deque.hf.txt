module std.deque (Deque, empty, null, size, pushFront, pushBack, popFront, popBack, peekFront, peekBack, fromList, toList)

data Deque a
empty : {a : Type} -> Deque a
null : {a : Type} -> Deque a -> Bool
size : {a : Type} -> Deque a -> Int
pushFront : {a : Type} -> Deque a -> a -> Deque a
pushBack : {a : Type} -> Deque a -> a -> Deque a
popFront : {a : Type} -> Deque a -> Maybe (a, Deque a)
popBack : {a : Type} -> Deque a -> Maybe (a, Deque a)
peekFront : {a : Type} -> Deque a -> Maybe a
peekBack : {a : Type} -> Deque a -> Maybe a
fromList : {a : Type} -> List a -> Deque a
toList : {a : Type} -> Deque a -> List a
