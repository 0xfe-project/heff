module std.http.sse (Field(..), text, event, send, headers, reply)

import co
import bytes as Bytes
import std.http.server as Server
import stream as Stream

data Field = Event String | Data String | EventId String | Retry Int | Comment String
text : List Field -> String
event : List Field -> Bytes
send : {e : HFunctor} -> {{ys : Stream.yielding Bytes <: e}} -> List Field -> Co e Unit
headers : List (String, String)
reply : {e : HFunctor} -> List (String, String) -> Co (Stream.yielding Bytes <+> e) Unit -> Server.Reply e
