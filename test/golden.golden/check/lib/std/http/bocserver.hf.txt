module std.http.bocserver (serve, Responder, Answer(..), Streaming, Socket, Sender, send, finish, senderNumber)

import co
import exception
import resource
import boc as BOC
import bytes as Bytes
import std.duration as Duration
import std.http as HTTP
import std.http.server as Server
import std.http.sse as SSE
import std.http.websocket as WebSocket
import std.net as Net
import std.ordmap as OrdMap
import std.time as Time
import uint8 as UInt8

data Sender a
send : {f : HFunctor} -> {a : Type} -> {{bc : BOC.boc <: f}} -> Sender a -> a -> Co f Unit
finish : {f : HFunctor} -> {a : Type} -> {{bc : BOC.boc <: f}} -> Sender a -> Co f Unit
senderNumber : {a : Type} -> Sender a -> Int
type Streaming (e : HFunctor) err = { opened : Sender Bytes -> Co (exception err <+> e) Unit, gone : Sender Bytes -> Co (exception err <+> e) Unit }
type Socket (e : HFunctor) err = { opened : Sender WebSocket.Message -> Co (exception err <+> e) Unit, received : Sender WebSocket.Message -> WebSocket.Message -> Co (exception err <+> e) Unit, gone : Sender WebSocket.Message -> Co (exception err <+> e) Unit }
data Answer (e : HFunctor) err = Whole HTTP.Response | Streamed Server.Head (Streaming e err) | Upgraded (Socket e err)
type Responder (e : HFunctor) err = {f : HFunctor} -> {{bc : BOC.boc <: f}} -> Answer e err -> Co f Unit
serve : {e : HFunctor} -> {rg : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {pending : Int := 1048576} -> {heartbeat : Int := 15000} -> {ping : Int := 30000} -> (err : Type) -> {{bc : BOC.boc <: e}} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> With rg Net.TcpListener -> (Server.Request -> Responder e err -> Co (exception err <+> e) Unit) -> Co e Unit
