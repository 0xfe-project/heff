module std.http.server (Request, header, plain, parseHead, BodyIs(..), Head(..), Got(..), reason, responseBytes, keptAlive, After(..), serveConnection, serveUntil, serve, Threads, serveConcurrently)

import co
import exception
import resource
import bytes as Bytes
import std.http as HTTP
import std.net as Net
import uint8 as UInt8

type Request = { method : String, target : String, version : String, headers : List (String, String), body : Bytes, peer : Net.Address }
header : String -> Request -> Maybe String
plain : Int -> String -> HTTP.Response
parseHead : Net.Address -> String -> Maybe Request
data BodyIs = Length Int | Refuse Int
data Head = HeadAt Bytes Int | HeadTooLong | HeadEnded
data Got = Asked Request Bytes | Refused Int | Ended
reason : Int -> String
responseBytes : String -> Bool -> HTTP.Response -> Bytes
keptAlive : Request -> Bool
data After = Next Bytes | Done Bool
serveConnection : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {{nt : Net.net <: e}} -> (Request -> Bool) -> (Request -> Co e HTTP.Response) -> With s Net.TcpStream -> Co e Bool
serveUntil : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {{nt : Net.net <: e}} -> {{ex : exception Net.NetError <: e}} -> (Request -> Bool) -> With s Net.TcpListener -> (Request -> Co e HTTP.Response) -> Co e Unit
serve : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {{nt : Net.net <: e}} -> {{ex : exception Net.NetError <: e}} -> With s Net.TcpListener -> (Request -> Co e HTTP.Response) -> Co e Unit

theory Threads (e : HFunctor) where
  spawn : Co e Unit -> Co e Unit
  Signal : Type
  signal : Co e Signal
  give : Signal -> Co e Unit
  awaited : Signal -> Co e Unit

serveConcurrently : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {{th : Threads e}} -> {{nt : Net.net <: e}} -> With s Net.TcpListener -> (Request -> Co e HTTP.Response) -> Co e Unit
