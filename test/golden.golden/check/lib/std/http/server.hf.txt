module std.http.server (Request, header, plain, parseHead, BodyIs(..), Head(..), Got(..), requestIn, reason, responseBytes, chunk, streamedHead, takeoverHead, keptAlive, After(..), Taken, Reply(..), serveConnection, serveUntil, serve, serveConnectionReplies, serveRepliesUntil, serveReplies)

type Request = { method : String, target : String, version : String, headers : List (String, String), body : Bytes, peer : Net.Address }
header : String -> Request -> Maybe String
plain : Int -> String -> HTTP.Response
parseHead : Net.Address -> String -> Maybe Request
data BodyIs = Length Int | Refuse Int
data Head = HeadAt Bytes Int | HeadTooLong | HeadEnded
data Got = Asked Request Bytes | Refused Int | Ended
requestIn : {from : Int := 0} -> Net.Address -> Bytes -> Either (Maybe Request, Int) Got
reason : Int -> String
responseBytes : String -> Bool -> HTTP.Response -> Bytes
chunk : Bytes -> Bytes
streamedHead : String -> Bool -> Int -> List (String, String) -> Bytes
takeoverHead : Int -> List (String, String) -> Bytes
keptAlive : Request -> Bool
data After = Next Bytes | Done Bool
type Taken s = { request : Request, connection : With s Net.TcpStream, rest : Bytes }
data Reply (e : HFunctor) = Whole HTTP.Response | Streamed Int (List (String, String)) (Co (Stream.yielding Bytes <+> e) Unit) | TakenOver Int (List (String, String)) ({s : Type} -> Taken s -> Co e Unit)
serveConnection : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> (Request -> Bool) -> (Request -> Co e HTTP.Response) -> With s Net.TcpStream -> Co e Bool
serveConnectionReplies : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> {{ex : exception Net.NetError <: e}} -> (Request -> Bool) -> (Request -> Co e (Reply e)) -> With s Net.TcpStream -> Co e Bool
serveUntil : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> {{ex : exception Net.NetError <: e}} -> {{cu : Custodian.custodian <: e}} -> (Request -> Bool) -> With s Net.TcpListener -> (Request -> Co e HTTP.Response) -> Co e Unit
serve : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> {{ex : exception Net.NetError <: e}} -> {{cu : Custodian.custodian <: e}} -> With s Net.TcpListener -> (Request -> Co e HTTP.Response) -> Co e Unit
serveRepliesUntil : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> {{ex : exception Net.NetError <: e}} -> {{cu : Custodian.custodian <: e}} -> (Request -> Bool) -> With s Net.TcpListener -> (Request -> Co e (Reply e)) -> Co e Unit
serveReplies : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> {{ex : exception Net.NetError <: e}} -> {{cu : Custodian.custodian <: e}} -> With s Net.TcpListener -> (Request -> Co e (Reply e)) -> Co e Unit
