module std.http.router (Path, instanceCoeStringPath, instanceDivisionPath, int, string, matching, rest, instanceAdditionPath, query, queryIs, queryMatching, header, headerIs, headerMatching, Captured(..), Capture, instanceCaptureInt, instanceCaptureString, instanceCaptureListString, instanceCaptureMaybe, Run, Captures, instanceCapturesUnit, instanceCapturesInt, instanceCapturesString, instanceCapturesListString, instanceCapturesMaybe, instanceCapturesPair, Reply(..), Takes, instanceTakesResponse, instanceTakesAnswer, instanceTakesStated, instanceTakesLater, instanceTakesArrow, Kept, kept, Keeping, instanceKeepingCown, instanceKeepingKept, On, on, Handles, instanceHandlesRequest, instanceHandlesOn, Asked, Slot, Route, waiting, route, get, post, put, delete, patch, Routes, spread, RouteRegister, instanceRouteRegisterPair, Conflict(..), instanceShowConflict, Router, compiled, serving, dispatch)

import co
import exception
import array as Array
import bytes as Bytes
import boc as BOC
import int as Int
import regex as Regex
import std.hashmap as HashMap
import std.http as HTTP
import std.http.bocserver as BocServer
import stream as Stream

data Path
instance instanceCoeStringPath : Coe String Path
instance instanceDivisionPath : Division Path
instance instanceAdditionPath : Addition Path
int : Path
string : Path
matching : String -> Path
rest : Path
query : String -> Path
queryIs : String -> String -> Path
queryMatching : String -> String -> Path
header : String -> Path
headerIs : String -> String -> Path
headerMatching : String -> String -> Path
data Captured = One String | Many (List String) | Missing

theory Capture a where
  kind : String
  every : Bool
  read : Captured -> Maybe a

instance instanceCaptureInt : Capture Int
instance instanceCaptureString : Capture String
instance instanceCaptureListString : Capture (List String)
instance instanceCaptureMaybe : {a : Type} -> {{cp : Capture a}} -> Capture (Maybe a)
type Run (e : HFunctor) = Co (exception Empty <+> e) Unit

theory Captures a where
  kinds : List String
  always : List Bool
  decoded : List Captured -> Maybe a

instance instanceCapturesUnit : Captures Unit
instance instanceCapturesInt : Captures Int
instance instanceCapturesString : Captures String
instance instanceCapturesListString : Captures (List String)
instance instanceCapturesMaybe : {a : Type} -> {{cp : Capture a}} -> Captures (Maybe a)
instance instanceCapturesPair : {a b : Type} -> {{l : Captures a}} -> {{r : Capture b}} -> Captures (a, b)
data Reply (e : HFunctor) = Now (BocServer.Answer e Empty) | Later (Co e (BocServer.Answer e Empty))

theory Takes (e : HFunctor) s k where
  kinds : List String
  always : List Bool
  writes : Bool
  taken : List Captured -> Maybe (s -> k -> (s, Reply e))

instance instanceTakesResponse : {e : HFunctor} -> {s : Type} -> Takes e s HTTP.Response
instance instanceTakesAnswer : {e : HFunctor} -> {s : Type} -> Takes e s (BocServer.Answer e Empty)
instance instanceTakesStated : {e : HFunctor} -> {s : Type} -> Takes e s (s, HTTP.Response)
instance instanceTakesLater : {e : HFunctor} -> {s : Type} -> Takes e s (Co e (BocServer.Answer e Empty))
instance instanceTakesArrow : {e : HFunctor} -> {s a k : Type} -> {{cp : Capture a}} -> {{tk : Takes e s k}} -> Takes e s (a -> k)
data Kept (e : HFunctor) s
kept : {e : HFunctor} -> {s : Type} -> BOC.Cown s -> (s -> (s, Co e Unit)) -> Kept e s

theory Keeping (e : HFunctor) c s where
  keptOf : c -> Kept e s

instance instanceKeepingCown : {e : HFunctor} -> {s : Type} -> Keeping e (BOC.Cown s) s
instance instanceKeepingKept : {e : HFunctor} -> {s : Type} -> Keeping e (Kept e s) s
data On (e : HFunctor) s k
on : {e : HFunctor} -> {c s k : Type} -> {{kp : Keeping e c s}} -> c -> (s -> HTTP.ServerRequest -> k) -> On e s k
type Asked (e : HFunctor) a = { request : HTTP.ServerRequest, respond : BocServer.Responder e Empty, captured : a }
type Slot (e : HFunctor) a = BOC.Cown (Array (Asked e a))

theory Handles (e : HFunctor) h where
  kinds : List String
  always : List Bool
  reads : List Captured -> Bool
  opened : h -> Int -> Co e (Deliver e)

instance instanceHandlesRequest : {e : HFunctor} -> {k : Type} -> {{bc : BOC.boc <: e}} -> {{tk : Takes e Unit k}} -> Handles e (HTTP.ServerRequest -> k)
instance instanceHandlesOn : {e : HFunctor} -> {s k : Type} -> {{bc : BOC.boc <: e}} -> {{tk : Takes e s k}} -> Handles e (On e s k)
type Route (e : HFunctor) = { method : String, path : Path, kinds : List String, always : List Bool, reads : List Captured -> Bool, slots : Int, number : Int, open : Int -> Co e (Deliver e) }
waiting : {e : HFunctor} -> {f : HFunctor} -> {a : Type} -> {{bc : BOC.boc <: e}} -> {{cs : Captures a}} -> {{ys : Stream.yielding (Route e) <: f}} -> String -> Path -> (Slot e a -> Co e Unit) -> Co f Unit
route : {e : HFunctor} -> {f : HFunctor} -> {h : Type} -> {{hs : Handles e h}} -> {{ys : Stream.yielding (Route e) <: f}} -> String -> Path -> h -> Co f Unit
get : {e : HFunctor} -> {f : HFunctor} -> {h : Type} -> {{hs : Handles e h}} -> {{ys : Stream.yielding (Route e) <: f}} -> Path -> h -> Co f Unit
post : {e : HFunctor} -> {f : HFunctor} -> {h : Type} -> {{hs : Handles e h}} -> {{ys : Stream.yielding (Route e) <: f}} -> Path -> h -> Co f Unit
put : {e : HFunctor} -> {f : HFunctor} -> {h : Type} -> {{hs : Handles e h}} -> {{ys : Stream.yielding (Route e) <: f}} -> Path -> h -> Co f Unit
delete : {e : HFunctor} -> {f : HFunctor} -> {h : Type} -> {{hs : Handles e h}} -> {{ys : Stream.yielding (Route e) <: f}} -> Path -> h -> Co f Unit
patch : {e : HFunctor} -> {f : HFunctor} -> {h : Type} -> {{hs : Handles e h}} -> {{ys : Stream.yielding (Route e) <: f}} -> Path -> h -> Co f Unit
type Routes (e : HFunctor) = Co (Stream.yielding (Route e)) Unit

theory RouteRegister (e : HFunctor) s where
  routes : s -> Routes e

instance instanceRouteRegisterPair : {e : HFunctor} -> {s t : Type} -> {{rs : RouteRegister e s}} -> {{rt : RouteRegister e t}} -> RouteRegister e (s, t)
spread : {e : HFunctor} -> Int -> Routes e -> Routes e
data Conflict = Twice String String | Mismatch String String (List String) (List String) | RestNotLast String String | BadPattern String String String | Misjoined String String String
instance instanceShowConflict : Show Conflict
data Router (e : HFunctor)
compiled : {e : HFunctor} -> Routes e -> Co e (Either (List Conflict) (Router e))
serving : {e : HFunctor} -> {s : Type} -> {{rr : RouteRegister e s}} -> s -> Co e (Either (List Conflict) (Router e))
dispatch : {e : HFunctor} -> {{bc : BOC.boc <: e}} -> Router e -> (HTTP.ServerRequest -> HTTP.Response) -> HTTP.ServerRequest -> BocServer.Responder e Empty -> Run e
