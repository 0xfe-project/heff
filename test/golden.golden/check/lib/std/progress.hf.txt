import co
import std.process

data Task = MkTask Int
root : Task
data ProgressF (m : Type -> Type) x = Start Task String Int (Task -> x) | Complete Task x | End Task x | RunUnder Task String (List String) (Ran -> x)
progress : HFunctor
start : {e : HFunctor} -> {{ev : progress <: e}} -> Task -> String -> Int -> Co e Task
complete : {e : HFunctor} -> {{ev : progress <: e}} -> Task -> Co e Unit
end : {e : HFunctor} -> {{ev : progress <: e}} -> Task -> Co e Unit
runUnder : {e : HFunctor} -> {{ev : progress <: e}} -> Task -> String -> List String -> Co e Ran
type Shown = { task : Int, parent : Int, name : String, done : Int, total : Int }
countOf : Int -> Int -> String
counted : Shown -> String
under : Int -> List Shown -> List Shown
branch : Bool -> String
stem : Bool -> String
drawUnder : List Shown -> String -> List Shown -> List String
drawTop : List Shown -> List Shown -> List String
draw : List Shown -> List String
data Kid = MkKid String Int Int Int
kidTask : Int -> Int -> Int
kidsUnder : Int -> Int -> List Kid -> List Shown
rooted : Int -> Kid -> Shown -> Shown
graft : Int -> List Kid -> List Shown -> List Shown
type Tasks = { next : Int, going : List Shown }
bumpOne : Int -> Shown -> Shown
bump : Int -> List Shown -> List Shown
parentOf : Int -> List Shown -> Int
started : Int -> String -> Int -> Tasks -> Tasks
ended : Int -> Tasks -> Tasks
data TerminalF (m : Type -> Type) x = IsTerminal (Bool -> x) | ShowLines (List String) x | ShowRun String (List String) (List Kid -> List String) (Ran -> x)
terminal : HFunctor
isTerminal : {e : HFunctor} -> {{ev : terminal <: e}} -> Co e Bool
showLines : {e : HFunctor} -> {{ev : terminal <: e}} -> List String -> Co e Unit
showRun : {e : HFunctor} -> {{ev : terminal <: e}} -> String -> List String -> (List Kid -> List String) -> Co e Ran
quietly : {m : Type -> Type} -> {e : HFunctor} -> {a : Type} -> {{sys : process <: e}} -> ProgressF m a -> Co e a
silently : {e : HFunctor} -> {a : Type} -> {{sys : process <: e}} -> Co (progress <+> e) a -> Co e a
redrawn : {e : HFunctor} -> {a : Type} -> {{term : terminal <: e}} -> Tasks -> a -> Co e (Pair Tasks a)
drawnOp : {m : Type -> Type} -> {e : HFunctor} -> {a : Type} -> {{term : terminal <: e}} -> Tasks -> ProgressF m a -> Co e (Pair Tasks a)
drawing : (e : HFunctor) -> {{term : terminal <: e}} -> Handler (plus progress e)
drawn : {e : HFunctor} -> {a : Type} -> {{term : terminal <: e}} -> Co (progress <+> e) a -> Co e a
shown : {e : HFunctor} -> {a : Type} -> {{sys : process <: e}} -> {{term : terminal <: e}} -> Co (progress <+> e) a -> Co e a
