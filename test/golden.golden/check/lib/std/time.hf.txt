module std.time

import co

data TimeF (m : Type -> Type) x = Sleep Int x | Now (Int -> x) | Monotonic (Int -> x)
time : HFunctor
sleep : {e : HFunctor} -> {{ev : time <: e}} -> Int -> Co e Unit
now : {e : HFunctor} -> {{ev : time <: e}} -> Co e Int
monotonic : {e : HFunctor} -> {{ev : time <: e}} -> Co e Int
