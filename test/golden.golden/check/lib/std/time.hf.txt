module std.time

import co
import std.duration as Duration
import std.instant as Instant

data TimeF (m : Type -> Type) x = Sleep Int x | Now (Int -> x) | Monotonic (Int -> x)
time : HFunctor
sleep : {e : HFunctor} -> {{ev : time <: e}} -> Duration.Duration -> Co e Unit
now : {e : HFunctor} -> {{ev : time <: e}} -> Co e Instant.Instant
monotonic : {e : HFunctor} -> {{ev : time <: e}} -> Co e Duration.Duration
