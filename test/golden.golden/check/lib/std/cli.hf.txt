module std.cli (Complete(..), Flag, Arity(..), Pos, Shape, Problem(..), Args, Term, Reader, readerOf, shapeOf, switch, option, choose, arg, optional, some, many, rest, with1, with2, with3, with4, shapeProblem, longNamed, longMaybe, flagLike, distance, Split, At, Splitter, splitWords, groupsGo, parse, problemOf, Candidates(..), Where, complete, Command, command, hidden, parseCommand, shapeOfCommand, posWord, flagWord, synopsis, headOf, usage, flagLine, helpOf, problemSaid, findCommand, completeLine, chosenVariable, fishScript, bashScript, zshScript, Outcome(..), readLine, compileArgs, listArgs, runArgs, commandsEx)

import rune as Rune

data Complete = None | Files String | Dirs | Among (List String)
type Flag = { long : String, short : String, meta : String, doc : String, values : List String, repeated : Bool, required : Bool, complete : Complete }
data Arity = One | Optional | Many | Some | Rest
type Pos = { meta : String, arity : Arity, complete : Complete }
type Shape = { flags : List Flag, pos : List Pos }
data Problem = Unknown String (Maybe String) | NoValue String | SwitchValue String | Ambiguous String (List String) | MissingFlag String | NotIn String String (List String) | Missing String | Extra String | Help | BadShape String

theory Args where
  A : Type -> Type
  give : {a : Type} -> a -> A a
  ap : {a b : Type} -> A (a -> b) -> A a -> A b
  flag : Flag -> A (List String)
  pos : Pos -> A (List String)

type Term a = (s : Args) -> s.A a
type Reader a = { shape : Shape, take : List (List String) -> List (String, List String) -> a }
readerOf : {a : Type} -> Term a -> Reader a
shapeOf : {a : Type} -> Term a -> Shape
switch : (s : Args) -> String -> String -> String -> s.A Bool
option : (s : Args) -> String -> String -> String -> String -> s.A (Maybe String)
choose : (s : Args) -> String -> String -> String -> String -> List String -> s.A String
arg : (s : Args) -> String -> Complete -> s.A String
optional : (s : Args) -> String -> Complete -> s.A (Maybe String)
some : (s : Args) -> String -> Complete -> s.A (List String)
many : (s : Args) -> String -> Complete -> s.A (List String)
rest : (s : Args) -> String -> s.A (List String)
with1 : (s : Args) -> {a r : Type} -> (a -> r) -> s.A a -> s.A r
with2 : (s : Args) -> {a b r : Type} -> (a -> b -> r) -> s.A a -> s.A b -> s.A r
with3 : (s : Args) -> {a b c r : Type} -> (a -> b -> c -> r) -> s.A a -> s.A b -> s.A c -> s.A r
with4 : (s : Args) -> {a b c d r : Type} -> (a -> b -> c -> d -> r) -> s.A a -> s.A b -> s.A c -> s.A d -> s.A r
shapeProblem : Shape -> Maybe String
longNamed : Shape -> String -> Either (List String) Flag
longMaybe : Shape -> String -> Maybe Flag
flagLike : Shape -> String -> Bool
distance : String -> String -> Int
type Split = { pos : List String, flags : List (String, String) }
type At = { sh : Shape, ra : Int, on : Bool, ps : List String, fs : List (String, String) }
type Splitter = { words : At -> List String -> Either Problem Split }
splitWords : Shape -> List String -> Either Problem Split
groupsGo : List Pos -> List String -> Either Problem (List (List String))
parse : {a : Type} -> Term a -> List String -> Either Problem a
problemOf : Shape -> List String -> Maybe Problem
data Candidates = Words (List String) | FilesBy String | DirsOnly | Nothing
type Where = { on : Bool, waiting : Maybe Flag, count : Int, given : List String }
complete : Shape -> List String -> String -> Candidates
type Command r = { name : String, doc : String, shown : Bool, args : Term r }
command : {r : Type} -> String -> String -> Term r -> Command r
hidden : {r : Type} -> String -> Term r -> Command r
parseCommand : {r : Type} -> Command r -> List String -> Either Problem r
shapeOfCommand : {r : Type} -> Command r -> Shape
posWord : Pos -> String
flagWord : Flag -> String
synopsis : Shape -> String
headOf : {r : Type} -> Command r -> String
usage : {r : Type} -> String -> List (Command r) -> String
flagLine : Flag -> String
helpOf : {r : Type} -> String -> Command r -> String
problemSaid : Problem -> String
findCommand : {r : Type} -> List (Command r) -> String -> Maybe (Command r)
completeLine : {r : Type} -> List (Command r) -> List String -> String
chosenVariable : String -> String
fishScript : String -> String
bashScript : String -> String
zshScript : String -> String
data Outcome r = Run r | Out String | Fail String
readLine : {r : Type} -> String -> List (Command r) -> List String -> Outcome r
compileArgs : Term String
listArgs : Term String
runArgs : Term String
commandsEx : List (Command String)
