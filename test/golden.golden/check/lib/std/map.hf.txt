module std.map

import std.hashmap as HashMap
import std.hashtable as HashTable
import std.ordmap as OrdMap

theory Mapping c where
  Key : Type
  Value : Type
  empty : c
  insert : c -> Key -> Value -> c
  lookup : c -> Key -> Maybe Value
  remove : c -> Key -> (Maybe Value, c)
  size : c -> Int
  foldl : {b : Type} -> (b -> Key -> Value -> b) -> b -> c -> b

empty : {c : Type} -> {{m : Mapping c}} -> c
insert : {c : Type} -> {{m : Mapping c}} -> c -> m.Key -> m.Value -> c
lookup : {c : Type} -> {{m : Mapping c}} -> c -> m.Key -> Maybe m.Value
remove : {c : Type} -> {{m : Mapping c}} -> c -> m.Key -> (Maybe m.Value, c)
size : {c : Type} -> {{m : Mapping c}} -> c -> Int
foldl : {c b : Type} -> {{m : Mapping c}} -> (b -> m.Key -> m.Value -> b) -> b -> c -> b
instance instanceMappingHashMap : {k v : Type} -> {{hashing : HashMap.Hash k}} -> {{equality : Equality k}} -> Mapping (HashMap.HashMap k v)
instance instanceMappingHashTable : {k v : Type} -> {{hashing : HashMap.Hash k}} -> {{equality : Equality k}} -> Mapping (HashTable.HashTable k v)
instance instanceMappingOrdMap : {k v : Type} -> {{ord : Order k}} -> Mapping (OrdMap.OrdMap k v)
tally : {c : Type} -> {{m : Mapping c @ { Value = Int }}} -> List m.Key -> c
