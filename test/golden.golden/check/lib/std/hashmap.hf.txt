module std.hashmap (strHash, intHash, strHashRange, strEqRange, trieSlot, Hash, hash, instanceHashString, instanceHashInt, instanceHashRune, instanceHashBool, combine, instanceHashPair, instanceHashMaybe, instanceHashList, KeyView, Slice(..), slice, instanceKeyViewStringSlice, Trie(..), HashMap(..), empty, Walked(..), update, insert, lookup, remove, delete, alter, foldl, size, toList, member, keys, values, instanceIterateHashMap, fromList, updateBy, lookupBy)

import array as Array
import int as Int

strHash : String -> Int
intHash : Int -> Int
strHashRange : String -> Int -> Int -> Int
strEqRange : String -> String -> Int -> Int -> Bool
trieSlot : Int -> Int -> Int -> Int

theory Hash a where
  hash : a -> Int

hash : {a : Type} -> {{hashing : Hash a}} -> a -> Int
instance instanceHashString : Hash String
instance instanceHashInt : Hash Int
instance instanceHashRune : Hash Rune
instance instanceHashBool : Hash Bool
combine : Int -> Int -> Int
instance instanceHashPair : {a b : Type} -> {{hashA : Hash a}} -> {{hashB : Hash b}} -> Hash (a, b)
instance instanceHashMaybe : {a : Type} -> {{hashing : Hash a}} -> Hash (Maybe a)
instance instanceHashList : {a : Type} -> {{hashing : Hash a}} -> Hash (List a)

theory KeyView k q where
  hashOf : q -> Int
  matches : k -> q -> Bool
  keyOf : q -> k

data Slice = MkSlice String Int Int
slice : String -> Int -> Int -> Slice
instance instanceKeyViewStringSlice : KeyView String Slice
data Trie k v = Vacant | Leaf Int k v | Collision Int (List (k, v)) | Branch Int (Array (Trie k v))
data HashMap k v = MkHashMap Int (Trie k v)
empty : {k v : Type} -> HashMap k v
data Walked k v a = MkWalked (Trie k v) a (Array (Trie k v))
update : {k v : Type} -> {{hashing : Hash k}} -> {{equality : Equality k}} -> HashMap k v -> k -> v -> (v -> v -> v) -> HashMap k v
insert : {k v : Type} -> {{hashing : Hash k}} -> {{equality : Equality k}} -> HashMap k v -> k -> v -> HashMap k v
lookup : {k v : Type} -> {{hashing : Hash k}} -> {{equality : Equality k}} -> HashMap k v -> k -> Maybe v
remove : {k v : Type} -> {{hashing : Hash k}} -> {{equality : Equality k}} -> HashMap k v -> k -> (Maybe v, HashMap k v)
delete : {k v : Type} -> {{hashing : Hash k}} -> {{equality : Equality k}} -> HashMap k v -> k -> HashMap k v
alter : {k v : Type} -> {{hashing : Hash k}} -> {{equality : Equality k}} -> HashMap k v -> k -> (Maybe v -> Maybe v) -> HashMap k v
foldl : {k v b : Type} -> (b -> k -> v -> b) -> b -> HashMap k v -> b
size : {k v : Type} -> HashMap k v -> Int
toList : {k v : Type} -> HashMap k v -> List (k, v)
member : {k v : Type} -> {{hashing : Hash k}} -> {{equality : Equality k}} -> HashMap k v -> k -> Bool
keys : {k v : Type} -> HashMap k v -> List k
values : {k v : Type} -> HashMap k v -> List v
instance instanceIterateHashMap : {k v : Type} -> Iterate (HashMap k v)
fromList : {k v : Type} -> {{hashing : Hash k}} -> {{equality : Equality k}} -> List (k, v) -> HashMap k v
updateBy : {k v q : Type} -> {{view : KeyView k q}} -> HashMap k v -> q -> v -> (v -> v -> v) -> HashMap k v
lookupBy : {k v q : Type} -> {{view : KeyView k q}} -> HashMap k v -> q -> Maybe v
