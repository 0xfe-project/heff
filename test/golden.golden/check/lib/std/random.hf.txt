module std.random (Random, seeded, fromEntropy, split, next, below, between, float, shuffle, pick)

import co
import array as Array
import bytes as Bytes
import std.entropy as Entropy
import uint8 as UInt8
import uint64 as UInt64

data Random
next : Random -> (UInt64, Random)
seeded : Int -> Random
fromEntropy : {e : HFunctor} -> {{ev : Entropy.entropy <: e}} -> Co e Random
split : Random -> (Random, Random)
below : Random -> (bound : Int) -> (Int, Random)
between : Random -> Int -> Int -> (Int, Random)
float : Random -> (Float64, Random)
shuffle : {a : Type} -> Random -> List a -> (List a, Random)
pick : {a : Type} -> Random -> List a -> Maybe (a, Random)
