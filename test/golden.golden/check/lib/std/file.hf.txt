module std.file (Access(..), Exists(..), WriteOpts, From(..), open, close, withStdin, withStdout, withStderr, readChunk, readLine, writeBytes, writeString, seek, tell, stat, foldLines, foldLinesBy, foldStdinLines, copyOpen, instanceBinderOpen, instanceHolderOpen)

import co
import exception
import resource
import std.filesystem
import std.path
import bytes as Bytes

data Exists = Error | Truncate | MustTruncate | Replace | TruncateOrReplace | Update | CanUpdate | Append
type WriteOpts = { exists : Exists = Error, permissions : Int = 438 }
data Access = Read | Write WriteOpts | ReadWrite WriteOpts
data From = FromStart | FromCurrent | FromEnd
open : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> {access : Access := Read} -> Path -> Co e File
close : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> Co e Unit
withStdin : {e : HFunctor} -> {a : Type} -> {{ev : filesystem <: e}} -> (File -> Co e a) -> Co e a
withStdout : {e : HFunctor} -> {a : Type} -> {{ev : filesystem <: e}} -> (File -> Co e a) -> Co e a
withStderr : {e : HFunctor} -> {a : Type} -> {{ev : filesystem <: e}} -> (File -> Co e a) -> Co e a
readChunk : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> Int -> Co e Bytes
readLine : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> Co e (Maybe String)
writeBytes : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> Bytes -> Co e Unit
writeString : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> String -> Co e Unit
seek : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> From -> Int -> Co e Int
tell : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> Co e Int
stat : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> Co e Status
foldLinesBy : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> {a : Type} -> Int -> File -> a -> (a -> String -> Co e a) -> Co e a
foldLines : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> {a : Type} -> File -> a -> (a -> String -> Co e a) -> Co e a
foldStdinLines : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> {a : Type} -> a -> (a -> String -> Co e a) -> Co e a
copyOpen : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> File -> File -> Co e Int
instance instanceBinderOpen : {e : HFunctor} -> Binder (Co e File)
instance instanceHolderOpen : {e : HFunctor} -> {{ev : filesystem <: e}} -> {{ex : exception FileError <: e}} -> Holder e (Co e File) (\s -> File) (\x -> x) (\r -> r)
