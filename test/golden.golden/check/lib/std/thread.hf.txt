module std.thread

import co
import parallel
import ffi as FFI

type Work x = (r : Type) -> ((t : Type) -> (a : Type) -> (t -> a) -> List t -> (List a -> x) -> r) -> r
type Joined x = (r : Type) -> ((a : Type) -> a -> (List a -> x) -> r) -> r
data ThreadsF (m : Type -> Type) x = Pool Int (Work x) | Spread Int (Int -> x) | Gather (Joined x) | Cores (Int -> x)
work : {t a x : Type} -> (t -> a) -> List t -> (List a -> x) -> Work x
joined : {a x : Type} -> a -> (List a -> x) -> Joined x
workThen : {x y : Type} -> (x -> y) -> Work x -> Work y
joinedThen : {x y : Type} -> (x -> y) -> Joined x -> Joined y
threads : HFunctor
parMap : {e : HFunctor} -> {{th : threads <: e}} -> {b a : Type} -> Int -> (b -> a) -> List b -> Co e (List a)
parFor : {e : HFunctor} -> {{th : threads <: e}} -> {a : Type} -> Int -> Int -> (Int -> a) -> Co e (List a)
cores : {e : HFunctor} -> {{th : threads <: e}} -> Co e Int
threadsInC : FFI.Described
spread : {e : HFunctor} -> {{th : threads <: e}} -> Int -> Co e Int
gather : {e : HFunctor} -> {{th : threads <: e}} -> {a : Type} -> a -> Co e (List a)
bothSides : {a b x : Type} -> (a -> b -> x) -> List (Either a b) -> x
threaded : {e : HFunctor} -> {{th : threads <: e}} -> {a b x : Type} -> ParallelF a b (Co e) x -> Co e x
inThreads : (a : Type) -> (b : Type) -> (e : HFunctor) -> {{th : threads <: e}} -> Handler (parallel a b <+> e)
