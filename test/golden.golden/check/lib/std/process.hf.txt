import co

data Ended = Exited Int | Signalled Int | TimedOut | NotRun
data Input = NoInput | Given String | Piped | Inherited
data Output = Collected | Piped | Inherited
data Portion = Available | Exactly Int | Through String
data Received = Got Bytes | Timeout | Closed
type Invocation = { command : String, args : List String = [], dir : String = "", env : List (Pair String String) = [], input : Input = NoInput, output : Output = Collected, errors : Output = Collected, timeout : Int = 0, keep : Int = 0, group : Bool = False }
type Ran = { ended : Ended, out : String, err : String }
succeeded : Ran -> Bool
data Started
type ProcessError = { code : Int, name : String, message : String }

setWorkingDir : {e : HFunctor} -> {{ev : process <: e}} -> String -> Co e Bool
platform : {e : HFunctor} -> {{ev : process <: e}} -> Co e String
processors : {e : HFunctor} -> {{ev : process <: e}} -> Co e Int
exit : {e : HFunctor} -> {{ev : process <: e}} -> Int -> Co e Unit

run : {e : HFunctor} -> {{ev : process <: e}} -> Invocation -> Co e Ran

terminate : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Co e Ran
writeInput : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Bytes -> Co e Bool
closeInput : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Co e Unit
readOutput : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Portion -> Int -> Co e Received
