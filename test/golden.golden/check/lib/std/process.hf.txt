module std.process (Ended(..), Input(..), Output(..), Portion(..), Received(..), Invocation, Ran, succeeded, Started, ProcessError, instanceShowProcessError, process, args, getEnv, programPath, workingDir, setWorkingDir, platform, processors, exit, exec, run, withProcess, wait, tryWait, kill, terminate, writeInput, closeInput, readOutput, readErrors)

import co
import exception
import resource
import std.path

type ProcessError = { code : Int, name : String, message : String, path : String }
instance instanceShowProcessError : Show ProcessError
data Ended = Exited Int | Signalled Int | TimedOut | NotRun ProcessError
data Input = NoInput | Given String | Piped | Inherited
data Output = Collected | Piped | Inherited
data Portion = Available | Exactly Int | Through String
data Received = Got Bytes | Timeout | Closed
type Invocation = { command : String, args : List String = [], dir : Path = current, env : List (String, String) = [], input : Input = NoInput, output : Output = Collected, errors : Output = Collected, timeout : Int = 0, keep : Int = 0, group : Bool = False }
type Ran = { ended : Ended, out : String, err : String }
succeeded : Ran -> Bool
data Started
process : HFunctor

workingDir : {e : HFunctor} -> {{ev : process <: e}} -> Co e Path
setWorkingDir : {e : HFunctor} -> {{ev : process <: e}} -> Path -> Co e Bool
platform : {e : HFunctor} -> {{ev : process <: e}} -> Co e String
processors : {e : HFunctor} -> {{ev : process <: e}} -> Co e Int
exit : {e : HFunctor} -> {{ev : process <: e}} -> Int -> Co e Unit
exec : {e : HFunctor} -> {{ev : process <: e}} -> String -> List String -> Co e Unit
run : {e : HFunctor} -> {{ev : process <: e}} -> Invocation -> Co e Ran
withProcess : {e : HFunctor} -> {a : Type} -> {{ev : process <: e}} -> {{ex : exception ProcessError <: e}} -> Invocation -> ({s : Type} -> With s Started -> Co e a) -> Co e a

tryWait : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Co e (Maybe Ran)
kill : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Co e Unit
terminate : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Co e Ran
writeInput : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Bytes -> Co e Bool
closeInput : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Co e Unit
readOutput : {e : HFunctor} -> {s : Type} -> {{ev : process <: e}} -> With s Started -> Portion -> Int -> Co e Received
