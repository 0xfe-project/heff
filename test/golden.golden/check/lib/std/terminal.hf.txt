import co
import std.terminal.widget as Widget

data ConsoleF (m : Type -> Type) x = Interactive (Bool -> x) | Begin (Bool -> x) | Finish x | Measure (Widget.Size -> x) | ReadInput (Maybe String -> x) | Draw String x
console : HFunctor
interactive : {e : HFunctor} -> {{ev : console <: e}} -> Co e Bool
begin : {e : HFunctor} -> {{ev : console <: e}} -> Co e Bool
finish : {e : HFunctor} -> {{ev : console <: e}} -> Co e Unit
measure : {e : HFunctor} -> {{ev : console <: e}} -> Co e Widget.Size
readInput : {e : HFunctor} -> {{ev : console <: e}} -> Co e (Maybe String)
draw : {e : HFunctor} -> {{ev : console <: e}} -> String -> Co e Unit
loop : {e : HFunctor} -> {{ev : console <: e}} -> {s r : Type} -> Widget.App s r -> s -> Co e (Maybe r)
interact : {e : HFunctor} -> {{ev : console <: e}} -> {s r : Type} -> Widget.App s r -> Co e (Maybe r)
