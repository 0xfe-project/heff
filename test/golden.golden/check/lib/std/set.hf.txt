module std.set

size : {a : Type} -> Set a -> Int
union : {a : Type} -> {{ord : Order a}} -> Set a -> Set a -> Set a
intersection : {a : Type} -> {{ord : Order a}} -> Set a -> Set a -> Set a
difference : {a : Type} -> {{ord : Order a}} -> Set a -> Set a -> Set a
isSubset : {a : Type} -> {{ord : Order a}} -> Set a -> Set a -> Bool
null : {a : Type} -> Set a -> Bool
