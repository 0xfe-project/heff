module std.net (TcpStream, TcpListener, Address, NetError, NetF, net, withConnection, read, write, shutdown, peer, thisSide, withListener, accept, accepted, close, closeListener, local, instanceShowNetError, showAddress)

import co
import exception
import resource

data TcpStream
data TcpListener
type Address = { host : String, port : Int }
type NetError = { code : Int, name : String, message : String, address : String }
data NetF (m : Type -> Type) x
net : HFunctor
withConnection : {e : HFunctor} -> {a : Type} -> {timeout : Int := 0} -> {{ev : net <: e}} -> {{ex : exception NetError <: e}} -> Address -> ({s : Type} -> With s TcpStream -> Co e a) -> Co e a
read : {e : HFunctor} -> {s : Type} -> {timeout : Int := 0} -> {{ev : net <: e}} -> With s TcpStream -> Int -> Co e (Either NetError Bytes)
write : {e : HFunctor} -> {s : Type} -> {timeout : Int := 0} -> {{ev : net <: e}} -> With s TcpStream -> Bytes -> Co e (Either NetError Unit)
shutdown : {e : HFunctor} -> {s : Type} -> {{ev : net <: e}} -> With s TcpStream -> Co e (Either NetError Unit)
peer : {e : HFunctor} -> {s : Type} -> {{ev : net <: e}} -> With s TcpStream -> Co e (Either NetError Address)
thisSide : {e : HFunctor} -> {s : Type} -> {{ev : net <: e}} -> With s TcpStream -> Co e (Either NetError Address)
withListener : {e : HFunctor} -> {a : Type} -> {{ev : net <: e}} -> Address -> Int -> ({s : Type} -> With s TcpListener -> Co e a) -> Co e (Either NetError a)
accept : {e : HFunctor} -> {s a : Type} -> {{ev : net <: e}} -> {{ex : exception NetError <: e}} -> With s TcpListener -> ({s2 : Type} -> With s2 TcpStream -> Co e a) -> Co e a
accepted : {e : HFunctor} -> {s : Type} -> {{ev : net <: e}} -> With s TcpListener -> Co e (Either NetError (With s TcpStream))
close : {e : HFunctor} -> {s : Type} -> {{ev : net <: e}} -> With s TcpStream -> Co e (Either NetError Unit)
closeListener : {e : HFunctor} -> {s : Type} -> {{ev : net <: e}} -> With s TcpListener -> Co e (Either NetError Unit)
local : {e : HFunctor} -> {s : Type} -> {{ev : net <: e}} -> With s TcpListener -> Co e (Either NetError Address)
instance instanceShowNetError : Show NetError
showAddress : Address -> String
