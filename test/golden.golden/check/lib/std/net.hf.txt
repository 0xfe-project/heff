import co
import std.handle

data TcpStream
data TcpListener
type Connection = Handle TcpStream
type Listener = Handle TcpListener
type Address = { host : String, port : Int }
type NetError = { code : Int, name : String, message : String }
data NetF (m : Type -> Type) x = Connect String Int (Either NetError Connection -> x) | Listen String Int Int (Either NetError Listener -> x) | Accept Listener (Either NetError Connection -> x) | Read Connection Int (Either NetError Bytes -> x) | Write Connection Bytes (Either NetError Unit -> x) | Shutdown Connection (Either NetError Unit -> x) | Close Int x | Peer Connection (Either NetError Address -> x) | Local Int (Either NetError Address -> x)
net : HFunctor
numberOf : {t : Type} -> Handle t -> Int
connect : {e : HFunctor} -> {{ev : net <: e}} -> String -> Int -> Co e (Either NetError Connection)
read : {e : HFunctor} -> {{ev : net <: e}} -> Connection -> Int -> Co e (Either NetError Bytes)
write : {e : HFunctor} -> {{ev : net <: e}} -> Connection -> Bytes -> Co e (Either NetError Unit)
shutdown : {e : HFunctor} -> {{ev : net <: e}} -> Connection -> Co e (Either NetError Unit)
peer : {e : HFunctor} -> {{ev : net <: e}} -> Connection -> Co e (Either NetError Address)
listen : {e : HFunctor} -> {{ev : net <: e}} -> String -> Int -> Int -> Co e (Either NetError Listener)
accept : {e : HFunctor} -> {{ev : net <: e}} -> Listener -> Co e (Either NetError Connection)
local : {e : HFunctor} -> {t : Type} -> {{ev : net <: e}} -> Handle t -> Co e (Either NetError Address)
close : {e : HFunctor} -> {t : Type} -> {{ev : net <: e}} -> Handle t -> Co e Unit
showError : NetError -> String
showAddress : Address -> String
