module std.crypto

import co
import ffi as FFI

data PrivateKey
data PublicKey
data Algorithm = RS256 | PS256 | ES256 | EdDSA
data CryptoF (m : Type -> Type) x = PrivatePem String (Either FFI.Errno PrivateKey -> x) | PublicPem String (Either FFI.Errno PublicKey -> x) | RsaJwk Bytes Bytes (Either FFI.Errno PublicKey -> x) | P256Jwk Bytes Bytes (Either FFI.Errno PublicKey -> x) | Ed25519Jwk Bytes (Either FFI.Errno PublicKey -> x) | PublicOf PrivateKey (PublicKey -> x) | PublicPemOf PublicKey (String -> x) | Sign Int PrivateKey Bytes (Either FFI.Errno Bytes -> x) | Verify Int PublicKey Bytes Bytes (Bool -> x) | Why (String -> x)
crypto : HFunctor
key : FFI.CType
cryptoInC : FFI.Described
cryptoLinks : FFI.Links
name : Algorithm -> String
named : String -> Maybe Algorithm
instance instanceShowAlgorithm : Show Algorithm
instance instanceEqualityAlgorithm : Equality Algorithm
codeOf : Algorithm -> Int
why : {e : HFunctor} -> {{ev : crypto <: e}} -> Co e String
said : {e : HFunctor} -> {a : Type} -> {{ev : crypto <: e}} -> Either FFI.Errno a -> Co e (Either String a)
privateKeyFromPem : {e : HFunctor} -> {{ev : crypto <: e}} -> String -> Co e (Either String PrivateKey)
publicKeyFromPem : {e : HFunctor} -> {{ev : crypto <: e}} -> String -> Co e (Either String PublicKey)
rsaPublicKey : {e : HFunctor} -> {{ev : crypto <: e}} -> Bytes -> Bytes -> Co e (Either String PublicKey)
p256PublicKey : {e : HFunctor} -> {{ev : crypto <: e}} -> Bytes -> Bytes -> Co e (Either String PublicKey)
ed25519PublicKey : {e : HFunctor} -> {{ev : crypto <: e}} -> Bytes -> Co e (Either String PublicKey)
publicOf : {e : HFunctor} -> {{ev : crypto <: e}} -> PrivateKey -> Co e PublicKey
sign : {e : HFunctor} -> {{ev : crypto <: e}} -> Algorithm -> PrivateKey -> Bytes -> Co e (Either String Bytes)
verify : {e : HFunctor} -> {{ev : crypto <: e}} -> Algorithm -> PublicKey -> Bytes -> Bytes -> Co e Bool
publicKeyToPem : {e : HFunctor} -> {{ev : crypto <: e}} -> PublicKey -> Co e String
