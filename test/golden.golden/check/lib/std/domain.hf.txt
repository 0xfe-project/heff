module std.domain (domain, Domain, Atomic, Mutex, Condition, spawn, join, cores, newAtomic, get, set, cas, fetchAdd, newMutex, lock, unlock, newCondition, wait, signal, broadcast)

import co

data Domain a
data Atomic a
data Mutex
data Condition
domain : HFunctor
spawn : {e : HFunctor} -> {a : Type} -> {{dm : domain <: e}} -> Co e a -> Co e (Domain a)
join : {e : HFunctor} -> {a : Type} -> {{dm : domain <: e}} -> Domain a -> Co e a
cores : {e : HFunctor} -> {{dm : domain <: e}} -> Co e Int
newAtomic : {e : HFunctor} -> {a : Type} -> {{dm : domain <: e}} -> a -> Co e (Atomic a)
get : {e : HFunctor} -> {a : Type} -> {{dm : domain <: e}} -> Atomic a -> Co e a
set : {e : HFunctor} -> {a : Type} -> {{dm : domain <: e}} -> Atomic a -> a -> Co e Unit
cas : {e : HFunctor} -> {a : Type} -> {{dm : domain <: e}} -> Atomic a -> a -> a -> Co e Bool
fetchAdd : {e : HFunctor} -> {{dm : domain <: e}} -> Atomic Int -> Int -> Co e Int
newMutex : {e : HFunctor} -> {{dm : domain <: e}} -> Co e Mutex
lock : {e : HFunctor} -> {{dm : domain <: e}} -> Mutex -> Co e Unit
unlock : {e : HFunctor} -> {{dm : domain <: e}} -> Mutex -> Co e Unit
newCondition : {e : HFunctor} -> {{dm : domain <: e}} -> Co e Condition
wait : {e : HFunctor} -> {{dm : domain <: e}} -> Condition -> Mutex -> Co e Unit
signal : {e : HFunctor} -> {{dm : domain <: e}} -> Condition -> Co e Unit
broadcast : {e : HFunctor} -> {{dm : domain <: e}} -> Condition -> Co e Unit
