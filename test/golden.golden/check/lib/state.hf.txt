import co

data StateF s (m : Type -> Type) x = Get (s -> x) | Put s x
state : (s : Type) -> HFunctor
get : {s : Type} -> {e : HFunctor} -> {{ev : state s <: e}} -> Co e s
put : {s : Type} -> {e : HFunctor} -> {{ev : state s <: e}} -> s -> Co e Unit
statefulFunctor : (s : Type) -> (e : HFunctor) -> RawFunctor
stateful : (s : Type) -> (e : HFunctor) -> Handler (plus (state s) e)
runState : {s : Type} -> {e : HFunctor} -> {a : Type} -> s -> Co (plus (state s) e) a -> Co e (Pair a s)
