module gpu (Device, Buffer, Pipeline, GpuError, GpuF(..), gpu, withDevice, withBuffer, writeBuffer, readBuffer, load, dispatch1, dispatch2, dispatch3, showError)

import co
import exception
import resource

data Device
data Buffer
data Pipeline
type GpuError = { message : String }
data GpuF (m : Type -> Type) x = Open (Either GpuError Device -> x) | Close Device (Either GpuError Unit -> x) | NewBuffer Device Int (Either GpuError Buffer -> x) | FreeBuffer Buffer (Either GpuError Unit -> x) | WriteBuffer Device Buffer Int Bytes (Either GpuError Unit -> x) | ReadBuffer Device Buffer Int Int (Either GpuError Bytes -> x) | Load Device String String (Either GpuError Pipeline -> x) | Dispatch Device Pipeline (List Buffer) Int Int Int (Either GpuError Unit -> x)
gpu : HFunctor
withDevice : {e : HFunctor} -> {a : Type} -> {{ev : gpu <: e}} -> {{ex : exception GpuError <: e}} -> ({s : Type} -> With s Device -> Co e a) -> Co e a
withBuffer : {e : HFunctor} -> {s a : Type} -> {{ev : gpu <: e}} -> {{ex : exception GpuError <: e}} -> With s Device -> Int -> ({t : Type} -> With t Buffer -> Co e a) -> Co e a
writeBuffer : {e : HFunctor} -> {s t : Type} -> {{ev : gpu <: e}} -> With s Device -> With t Buffer -> Int -> Bytes -> Co e (Either GpuError Unit)
readBuffer : {e : HFunctor} -> {s t : Type} -> {{ev : gpu <: e}} -> With s Device -> With t Buffer -> Int -> Int -> Co e (Either GpuError Bytes)
load : {e : HFunctor} -> {s : Type} -> {{ev : gpu <: e}} -> With s Device -> String -> String -> Co e (Either GpuError Pipeline)
dispatch1 : {e : HFunctor} -> {s t : Type} -> {{ev : gpu <: e}} -> With s Device -> Pipeline -> With t Buffer -> Int -> Int -> Int -> Co e (Either GpuError Unit)
dispatch2 : {e : HFunctor} -> {s t u : Type} -> {{ev : gpu <: e}} -> With s Device -> Pipeline -> With t Buffer -> With u Buffer -> Int -> Int -> Int -> Co e (Either GpuError Unit)
dispatch3 : {e : HFunctor} -> {s t u v : Type} -> {{ev : gpu <: e}} -> With s Device -> Pipeline -> With t Buffer -> With u Buffer -> With v Buffer -> Int -> Int -> Int -> Co e (Either GpuError Unit)
showError : GpuError -> String
