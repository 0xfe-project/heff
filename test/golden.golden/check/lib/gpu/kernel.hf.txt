data BinaryOp = Plus | Minus | Times | Quotient | Modulo | Less | AtMost | Greater | AtLeast | EqualTo | NotEqualTo | AndAlso | OrElse
data Expr = Literal String | Var String | Binary BinaryOp Expr Expr | Negated Expr | Inverted Expr | Index String Expr | Length String | Convert String Expr
data Exp t = UnsafeExp Expr
exprOf : {t : Type} -> Exp t -> Expr

theory Scalar a where
  typeName : String
  literal : a -> String

theory Numeric a where
  bytes : Int

theory Signed a where
  signedName : String

theory Integral a where
  integralName : String

data Read
data Write
data Buf m a = UnsafeBuf String
type Input a = Buf Read a
type Output a = Buf Write a

theory Mode m where
  access : String

data StmtF (m : Type -> Type) x = Let Expr (Expr -> x) | Store String Expr Expr (Unit -> x) | If Expr (m Unit) (m Unit) (Unit -> x) | For Expr Expr (Expr -> m Unit) (Unit -> x) | GlobalId (Expr -> x)
stmtMap : {m : Type -> Type} -> {a b : Type} -> (a -> b) -> StmtF m a -> StmtF m b
stmtThrough : {m : Type -> Type} -> {n : Type -> Type} -> ({x : Type} -> m x -> n x) -> {a : Type} -> StmtF m a -> StmtF n a
stmt : HFunctor
define : {a : Type} -> Exp a -> Co stmt (Exp a)

type Param = { name : String, access : String, element : String }
type Kernel = { entry : String, workgroup : Int, params : List Param, body : Co stmt Unit }

theory Parameter p where
  declared : String -> Param
  named : String -> p
