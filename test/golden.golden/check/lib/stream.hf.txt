module stream (YieldF(..), yielding, yield, Step(..), Source, source, fold, map, take, zip, each)

import co

data YieldF o (m : Type -> Type) x = Yield o x
yielding : (o : Type) -> HFunctor
yield : {o : Type} -> {e : HFunctor} -> {{ev : yielding o <: e}} -> o -> Co e Unit
data Step (e : HFunctor) o r = Done r | Next o (Co e (Step e o r))
type Source (e : HFunctor) o r = Co e (Step e o r)
source : {o : Type} -> {e : HFunctor} -> {r : Type} -> Co (yielding o <+> e) r -> Source e o r
fold : {e : HFunctor} -> {o r a : Type} -> Source e o r -> a -> (a -> o -> Co e a) -> Co e (a, r)
map : {e : HFunctor} -> {a b r : Type} -> (a -> b) -> Source e a r -> Source e b r
take : {e : HFunctor} -> {o r : Type} -> Int -> Source e o r -> Source e o (Source e o r)
zip : {e : HFunctor} -> {a b r q : Type} -> Source e a r -> Source e b q -> Source e (a, b) Unit
each : {e : HFunctor} -> {o r : Type} -> {{ev : yielding o <: e}} -> Source e o r -> Co e r
