module remote

import co
import exception
import bytes as Bytes
import int as Int
import uint8 as UInt8
import wire as Wire

data Operation a b = MkOperation Int String
operation : {a b : Type} -> Int -> String -> Operation a b
type Call x = { index : Int, argument : Bytes, answer : Bytes -> Maybe x }
givenTo : {b x : Type} -> (b -> x) -> Maybe b -> Maybe x
call : {a b x : Type} -> {{takes : Wire.Wire a}} -> {{gives : Wire.Wire b}} -> Operation a b -> a -> (b -> x) -> Call x
type Served x = { index : Int, signature : String, run : Bytes -> Maybe x }
madeOf : {a b x : Type} -> (a -> (b -> Bytes) -> x) -> (b -> Bytes) -> Maybe a -> Maybe x
serve : {a b x : Type} -> {{takes : Wire.Wire a}} -> {{gives : Wire.Wire b}} -> Operation a b -> (a -> (b -> Bytes) -> x) -> Served x

theory Api (h : HFunctor) where
  name : String
  fingerprint : Int
  calls : {m : Type -> Type} -> {x : Type} -> h.f m x -> Call x
  serves : {m : Type -> Type} -> List (Served (h.f m Bytes))

hashStep : Int -> UInt8 -> Int
fnv : String -> Int
fingerprintOf : String -> List String -> Int
api : (h : HFunctor) -> String -> ({m : Type -> Type} -> {x : Type} -> h.f m x -> Call x) -> ({m : Type -> Type} -> List (Served (h.f m Bytes))) -> Api h
signatures : {h : HFunctor} -> Api h -> List String
version : Int
data Kind = Asked | Answered | Failed
kindNumber : Kind -> Int
kindOf : Int -> Maybe Kind
type Frame = { kind : Kind, fingerprint : Int, index : Int, payload : Bytes }
frame : Kind -> Int -> Int -> Bytes -> Bytes
framed : Int -> Int -> Bytes -> Maybe Kind -> Either String Frame
readFrame : Bytes -> Either String Frame
data RemoteError = Unreachable String | Mismatched String | Refused String | Malformed String
showRemoteError : RemoteError -> String
hexDigits : Int -> Int -> String -> String
named : String -> Int -> String
failureOf : Int -> String -> RemoteError
saidIn : Bytes -> String
answerIn : {x : Type} -> Int -> Call x -> Kind -> Frame -> Either RemoteError x
answerOf : {x : Type} -> Int -> Call x -> Either String Frame -> Either RemoteError x
answerFrom : {x : Type} -> Int -> Call x -> Either RemoteError Bytes -> Either RemoteError x
type Transport (e : HFunctor) = { exchange : Bytes -> Co e (Either RemoteError Bytes) }
thrownOr : {e : HFunctor} -> {{failed : exception RemoteError <: e}} -> {x : Type} -> Either RemoteError x -> Co e x
sent : {h : HFunctor} -> {e : HFunctor} -> {{failed : exception RemoteError <: e}} -> Api h -> Transport e -> {a : Type} -> h.f (Co e) a -> Co e a
remotely : (h : HFunctor) -> (e : HFunctor) -> {{failed : exception RemoteError <: e}} -> Api h -> Transport e -> {a : Type} -> Co (h <+> e) a -> Co e a
failure : {h : HFunctor} -> Api h -> Int -> String -> Bytes
servedAt : {x : Type} -> Int -> List (Served x) -> Maybe (Served x)
done : {h : HFunctor} -> {e : HFunctor} -> Api h -> ({a : Type} -> h.f (Co e) a -> Co e a) -> Int -> Maybe (h.f (Co e) Bytes) -> Co e Bytes
servedFor : {h : HFunctor} -> {e : HFunctor} -> Api h -> ({a : Type} -> h.f (Co e) a -> Co e a) -> Frame -> Maybe (Served (h.f (Co e) Bytes)) -> Co e Bytes
respondTo : {h : HFunctor} -> {e : HFunctor} -> Api h -> ({a : Type} -> h.f (Co e) a -> Co e a) -> Either String Frame -> Co e Bytes
respond : {h : HFunctor} -> {e : HFunctor} -> Api h -> ({a : Type} -> h.f (Co e) a -> Co e a) -> Bytes -> Co e Bytes
loopback : {h : HFunctor} -> {e : HFunctor} -> Api h -> ({a : Type} -> h.f (Co e) a -> Co e a) -> Transport e
locally : (h : HFunctor) -> (e : HFunctor) -> {{failed : exception RemoteError <: e}} -> Api h -> ({a : Type} -> h.f (Co e) a -> Co e a) -> {a : Type} -> Co (h <+> e) a -> Co e a
