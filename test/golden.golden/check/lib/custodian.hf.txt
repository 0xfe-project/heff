module custodian (custodian, CustodianF(..), Custodian, custodianThen, custodianBy, make, under, manage, shutdown, failed, Asking(..), Ticket, kept, dropped, untended)

import co

data Custodian
data Ticket
data Asking x = Made (Maybe Custodian) (Custodian -> x) | ShutDown Custodian (List String -> x) | Dropped Ticket x | Failed String x
data CustodianF (m : Type -> Type) x = Asked (Asking x) | Managed Custodian (Unit -> m Unit) (Bool -> x) | Kept (Unit -> m Unit) (Ticket -> x)
custodianThen : {m : Type -> Type} -> {x y : Type} -> (x -> y) -> CustodianF m x -> CustodianF m y
custodianBy : {m : Type -> Type} -> {n : Type -> Type} -> {x : Type} -> ({t : Type} -> m t -> n t) -> CustodianF m x -> CustodianF n x
custodian : HFunctor
make : {e : HFunctor} -> {{cu : custodian <: e}} -> Co e Custodian
under : {e : HFunctor} -> {{cu : custodian <: e}} -> Custodian -> Co e Custodian
manage : {e : HFunctor} -> {{cu : custodian <: e}} -> Custodian -> Co e Unit -> Co e Bool
shutdown : {e : HFunctor} -> {{cu : custodian <: e}} -> Custodian -> Co e (List String)
failed : {e : HFunctor} -> {{cu : custodian <: e}} -> String -> Co e Unit
kept : {e : HFunctor} -> {{cu : custodian <: e}} -> Co e Unit -> Co e Ticket
dropped : {e : HFunctor} -> {{cu : custodian <: e}} -> Ticket -> Co e Unit
untended : {e : HFunctor} -> {a : Type} -> Co (custodian <+> e) a -> Co e a
