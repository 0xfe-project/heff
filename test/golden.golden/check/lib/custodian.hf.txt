module custodian (custodian, CustodianF(..), Asking(..), Giving(..), Custodian, Owned(..), custodianThen, custodianBy, make, under, manage, shutdown, owned, failed, current, custodied, Running(..), Custody, custody, contained, instanceBinderCustody, instanceHolderCustody, Ticket, kept, dropped, untended)

import co

data Custodian
data Ticket
data Owned a = MkOwned a
data Running (m : Type -> Type) x a = MkRunning (m a) (a -> x)
data Asking x = Made (Maybe Custodian) Bool (Custodian -> x) | ShutDown Custodian (List String -> x) | Dropped Ticket x | Own (Some Owned) x | Failed String x | Current (Maybe Custodian -> x)
data Giving (m : Type -> Type) x = Managed Custodian (Unit -> m Unit) (Bool -> x) | Kept (Unit -> m Unit) (Ticket -> x) | Under Custodian (Some (Running m x))
data CustodianF (m : Type -> Type) x = Asked (Asking x) | Given (Giving m x)
custodianThen : {m : Type -> Type} -> {x y : Type} -> (x -> y) -> CustodianF m x -> CustodianF m y
custodianBy : {m : Type -> Type} -> {n : Type -> Type} -> {x : Type} -> ({t : Type} -> m t -> n t) -> CustodianF m x -> CustodianF n x
custodian : HFunctor
make : {e : HFunctor} -> {{cu : custodian <: e}} -> Co e Custodian
under : {e : HFunctor} -> {{cu : custodian <: e}} -> Custodian -> Co e Custodian
manage : {e : HFunctor} -> {{cu : custodian <: e}} -> Custodian -> Co e Unit -> Co e Bool
shutdown : {e : HFunctor} -> {{cu : custodian <: e}} -> Custodian -> Co e (List String)
owned : {e : HFunctor} -> {{cu : custodian <: e}} -> {a : Type} -> a -> Co e Unit
failed : {e : HFunctor} -> {{cu : custodian <: e}} -> String -> Co e Unit
current : {e : HFunctor} -> {{cu : custodian <: e}} -> Co e (Maybe Custodian)
custodied : {e : HFunctor} -> {a : Type} -> {{cu : custodian <: e}} -> Custodian -> Co e a -> Co e a
kept : {e : HFunctor} -> {{cu : custodian <: e}} -> Co e Unit -> Co e Ticket
dropped : {e : HFunctor} -> {{cu : custodian <: e}} -> Ticket -> Co e Unit
data Custody
custody : Custody
contained : Custody
instance instanceBinderCustody : Binder Custody
instance instanceHolderCustody : {e : HFunctor} -> {{cu : custodian <: e}} -> Holder e Custody (\s -> Custodian) (\w -> w) (\r -> r)
untended : {e : HFunctor} -> {a : Type} -> Co (custodian <+> e) a -> Co e a
