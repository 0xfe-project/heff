module boc (Cown, Cowns, instanceCownsCown, instanceCownsPair, instanceBinderCown, instanceBinderCowns, instanceBinderCownsMore, instanceHolderCown, instanceHolderCowns, instanceHolderCownsMore, Await, await, instanceBinderAwait, instanceBinderCownsAwait, instanceHolderAwait, instanceHolderCownsAwait, boc, newCown, when, scheduled, scheduledWithReactor)

import co
import custodian as Custodian
import std.domain as Domain
import std.reactor as Reactor

data Cown (w : HFunctor) a

theory Cowns c where
  World : HFunctor
  Contents : Type
  width : Int
  handles : c -> List (Handle World)
  read : (Domain.domain <: World) -> c -> Co World Contents
  write : (Domain.domain <: World) -> c -> List Bool -> Contents -> Co World Unit

instance instanceCownsCown : {w : HFunctor} -> {a : Type} -> Cowns (Cown w a)
instance instanceCownsPair : {c d : Type} -> {{l : Cowns c}} -> {{r : Cowns d @ { World = l.World }}} -> Cowns (c, d)
boc : HFunctor -> HFunctor
newCown : {e : HFunctor} -> {w : HFunctor} -> {a : Type} -> {{bc : boc w <: e}} -> a -> Co e (Cown w a)
when : {c : Type} -> {e : HFunctor} -> c -> {{cw : Cowns c}} -> {{bc : boc cw.World <: e}} -> (cw.Contents -> Co (boc cw.World <+> cw.World) cw.Contents) -> Co e Unit
instance instanceBinderCown : {w : HFunctor} -> {a : Type} -> Binder (Cown w a)
instance instanceBinderCowns : {w : HFunctor} -> {a b : Type} -> {{cw : Cowns (Cown w a, Cown w b)}} -> Binder (Cown w a, Cown w b)
instance instanceBinderCownsMore : {w : HFunctor} -> {a d f : Type} -> {{cw : Cowns (Cown w a, (d, f))}} -> Binder (Cown w a, (d, f))
instance instanceHolderCown : {e : HFunctor} -> {w : HFunctor} -> {a : Type} -> {{bc : boc w <: e}} -> Holder e (Cown w a) (\s -> a) (\x -> boc w <+> w) (\r -> Unit)
instance instanceHolderCowns : {e : HFunctor} -> {w : HFunctor} -> {a b : Type} -> {{cw : Cowns (Cown w a, Cown w b)}} -> {{bc : boc cw.World <: e}} -> Holder e (Cown w a, Cown w b) (\s -> cw.Contents) (\x -> boc cw.World <+> cw.World) (\r -> Unit)
instance instanceHolderCownsMore : {e : HFunctor} -> {w : HFunctor} -> {a d f : Type} -> {{cw : Cowns (Cown w a, (d, f))}} -> {{bc : boc cw.World <: e}} -> Holder e (Cown w a, (d, f)) (\s -> cw.Contents) (\x -> boc cw.World <+> cw.World) (\r -> Unit)
data Await (w : HFunctor) a
await : {w : HFunctor} -> {a : Type} -> Co w a -> Await w a
instance instanceBinderAwait : {w : HFunctor} -> {a : Type} -> Binder (Await w a)
instance instanceBinderCownsAwait : {c : Type} -> {w : HFunctor} -> {a : Type} -> {{cw : Cowns c @ { World = w }}} -> Binder (c, Await w a)
instance instanceHolderAwait : {e : HFunctor} -> {w : HFunctor} -> {a : Type} -> {{bc : boc w <: e}} -> Holder e (Await w a) (\s -> a) (\x -> boc w <+> w) (\r -> Unit)
instance instanceHolderCownsAwait : {e : HFunctor} -> {w : HFunctor} -> {c a : Type} -> {{cw : Cowns c @ { World = w }}} -> {{bc : boc w <: e}} -> Holder e (c, Await w a) (\s -> (cw.Contents, a)) (\x -> boc w <+> w) (\r -> Unit)
scheduled : {w : HFunctor} -> {a : Type} -> {batch : Int := 16} -> {{dm : Domain.domain <: w}} -> {{cu : Custodian.custodian <: w}} -> (String -> Co w Unit) -> Co (boc w <+> w) a -> Co w a
scheduledWithReactor : {w : HFunctor} -> {a : Type} -> {batch : Int := 16} -> {{dm : Domain.domain <: w}} -> {{cu : Custodian.custodian <: w}} -> {{rx : Reactor.reactor <: w}} -> (String -> Co w Unit) -> Co (boc w <+> w) a -> Co w a
