module boc (Cown, Cowns, instanceCownsCown, instanceCownsPair, instanceBinderCown, instanceBinderCowns, instanceHolderCown, instanceHolderCowns, boc, newCown, when, scheduled)

import co
import std.domain as Domain

data Cown (w : HFunctor) a

theory Cowns c where
  World : HFunctor
  Contents : Type
  width : Int
  handles : c -> List (Handle World)
  read : (Domain.domain <: World) -> c -> Co World Contents
  write : (Domain.domain <: World) -> c -> List Bool -> Contents -> Co World Unit

instance instanceCownsCown : {w : HFunctor} -> {a : Type} -> Cowns (Cown w a)
instance instanceCownsPair : {c d : Type} -> {{l : Cowns c}} -> {{r : Cowns d @ { World = l.World }}} -> Cowns (c, d)
boc : HFunctor -> HFunctor
newCown : {e : HFunctor} -> {w : HFunctor} -> {a : Type} -> {{bc : boc w <: e}} -> a -> Co e (Cown w a)
when : {c : Type} -> {e : HFunctor} -> c -> {{cw : Cowns c}} -> {{bc : boc cw.World <: e}} -> (cw.Contents -> Co e cw.Contents) -> Co e Unit
instance instanceBinderCown : {w : HFunctor} -> {a : Type} -> Binder (Cown w a)
instance instanceBinderCowns : {w : HFunctor} -> {a d : Type} -> {{cw : Cowns (Cown w a, d)}} -> Binder (Cown w a, d)
instance instanceHolderCown : {e : HFunctor} -> {w : HFunctor} -> {a : Type} -> {{bc : boc w <: e}} -> Holder e (Cown w a) (\s -> a) (\r -> Unit)
instance instanceHolderCowns : {e : HFunctor} -> {w : HFunctor} -> {a d : Type} -> {{cw : Cowns (Cown w a, d)}} -> {{bc : boc cw.World <: e}} -> Holder e (Cown w a, d) (\s -> cw.Contents) (\r -> Unit)
scheduled : {w : HFunctor} -> {a : Type} -> {{dm : Domain.domain <: w}} -> (String -> Co w Unit) -> Co (boc w <+> w) a -> Co w a
