module co

infixl <+>
infixl <: below <+>

theory RawFunctor where
  f : Type -> Type
  fmap : {a b : Type} -> (a -> b) -> f a -> f b

theory HFunctor where
  f : (Type -> Type) -> Type -> Type
  hfmap : (m : RawFunctor) -> {a b : Type} -> (a -> b) -> f m.f a -> f m.f b
  hmap : (m : RawFunctor) -> (n : RawFunctor) -> ({x : Type} -> m.f x -> n.f x) -> {a : Type} -> f m.f a -> f n.f a

theory Handler (h : HFunctor) where
  M : Type -> Type
  ret : {a : Type} -> a -> M a
  bind : {a b : Type} -> M a -> (a -> M b) -> M b
  malg : {a : Type} -> h.f M a -> M a

type Co (h : HFunctor) a = (hd : Handler h) -> (r : Type) -> (a -> hd.M r) -> hd.M r
val : {h : HFunctor} -> {a : Type} -> a -> Co h a
bind : {h : HFunctor} -> {a b : Type} -> Co h a -> (a -> Co h b) -> Co h b
pure : {h : HFunctor} -> {a : Type} -> a -> Co h a
instance instanceMonadCo : {h : HFunctor} -> Monad (Co h)
later : {h : HFunctor} -> {a : Type} -> (Unit -> a) -> Co h a
coFunctor : (h : HFunctor) -> RawFunctor
handlerFunctor : {h : HFunctor} -> Handler h -> RawFunctor
handle : {h : HFunctor} -> {a : Type} -> (hd : Handler h) -> Co h a -> hd.M a
op : {h : HFunctor} -> {a b : Type} -> h.f (Co h) a -> (a -> Co h b) -> Co h b
data SumF (f : (Type -> Type) -> Type -> Type) (g : (Type -> Type) -> Type -> Type) (m : Type -> Type) a = InL (f m a) | InR (g m a)
(<+>) : HFunctor -> HFunctor -> HFunctor
(<:) : HFunctor -> HFunctor -> Type
here : (h : HFunctor) -> h <: h
inl : (h : HFunctor) -> (k : HFunctor) -> h <: (h <+> k)
inr : (h : HFunctor) -> (k : HFunctor) -> k <: (h <+> k)
lift : {h : HFunctor} -> {k : HFunctor} -> {a : Type} -> {{inj : h <: k}} -> Co h a -> Co k a
through : (h : HFunctor) -> (e : HFunctor) -> ({a : Type} -> h.f (Co e) a -> Co e a) -> Handler (h <+> e)
extend : (k : HFunctor) -> (h : HFunctor) -> (hd : Handler h) -> ({a : Type} -> k.f hd.M a -> hd.M a) -> Handler (k <+> h)
within : (h : HFunctor) -> (e : HFunctor) -> ({a : Type} -> (hd : Handler e) -> h.f hd.M a -> hd.M a) -> {a : Type} -> Co (h <+> e) a -> Co e a

theory Binder c where
  Contents : Type -> Type
  World : HFunctor -> HFunctor = \w -> w
  Result : Type -> Type

theory Holder (e : HFunctor) c (contents : Type -> Type) (world : HFunctor -> HFunctor) (result : Type -> Type) where
  held : {r : Type} -> c -> List Bool -> ({s : Type} -> contents s -> Co (world e) (r, contents s)) -> Co e (result r)

holding : {e : HFunctor} -> {c r : Type} -> c -> {{b : Binder c}} -> {{h : Holder e c b.Contents b.World b.Result}} -> List Bool -> ({s : Type} -> b.Contents s -> Co (b.World e) (r, b.Contents s)) -> Co e (b.Result r)
