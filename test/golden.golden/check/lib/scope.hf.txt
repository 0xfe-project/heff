import co

data Armed
data Token = Numbered Int | Held Armed
data ScopeF (m : Type -> Type) x = Enter (m Unit) (Token -> x) | Leave Token x
scope : HFunctor
enter : {e : HFunctor} -> {{sc : scope <: e}} -> Co e Unit -> Co e Token
leave : {e : HFunctor} -> {{sc : scope <: e}} -> Token -> Co e Unit
scoped : {e : HFunctor} -> {r a : Type} -> {{sc : scope <: e}} -> Co e r -> (r -> Co e Unit) -> (r -> Co e a) -> Co e a
data Armory (e : HFunctor) = MkArmory Int (List (Pair Int (Co e Unit)))
disarmed : {e : HFunctor} -> Int -> List (Pair Int (Co e Unit)) -> List (Pair Int (Co e Unit))
tokenNumber : Token -> Int
armedWith : {e : HFunctor} -> {x : Type} -> Co e Unit -> (Token -> x) -> Armory e -> Pair (Armory e) x
leftOf : {e : HFunctor} -> Token -> Armory e -> Armory e
releasedAll : {e : HFunctor} -> List (Pair Int (Co e Unit)) -> Co e Unit
endedWith : {e : HFunctor} -> {a : Type} -> Pair (Armory e) a -> Co e a
armoredFunctor : (e : HFunctor) -> RawFunctor
armoryAlg : (e : HFunctor) -> {a : Type} -> ScopeF (\x -> Armory e -> Co e (Pair (Armory e) x)) a -> Armory e -> Co e (Pair (Armory e) a)
scopesHandler : (e : HFunctor) -> Handler (plus scope e)
endScopes : (e : HFunctor) -> {a : Type} -> Co (plus scope e) a -> Co e a
