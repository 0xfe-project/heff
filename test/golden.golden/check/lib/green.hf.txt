module green

import co
import std.ordmap as OrdMap

data GreenF (m : Type -> Type) x = Spawn (m Unit) x | Yield x | NewKey (Int -> x) | Wake Int Int x | Park Int (Int -> x)
green : HFunctor
spawn : {e : HFunctor} -> {{ev : green <: e}} -> Co e Unit -> Co e Unit
yield : {e : HFunctor} -> {{ev : green <: e}} -> Co e Unit
newKey : {e : HFunctor} -> {{ev : green <: e}} -> Co e Int
park : {e : HFunctor} -> {{ev : green <: e}} -> Int -> Co e Unit
wake : {e : HFunctor} -> {{ev : green <: e}} -> Int -> Co e Unit
newChan : {e : HFunctor} -> {{ev : green <: e}} -> Co e Int
send : {e : HFunctor} -> {{ev : green <: e}} -> Int -> Int -> Co e Unit
recv : {e : HFunctor} -> {{ev : green <: e}} -> Int -> Co e Int

theory Outside (e : HFunctor) where
  Group : Type -> Type
  group : {t : Type} -> Co e (Group t)
  start : {t : Type} -> Group t -> Co e t -> Co e (Maybe t)
  finished : {t : Type} -> Group t -> Bool -> Co e (List t)
  close : {t : Type} -> Group t -> Co e Unit

blocking : (e : HFunctor) -> Outside e
data Res (e : HFunctor) a = RDone a | RYield (Unit -> Res e a) | RSpawn (Res e Unit) (Unit -> Res e a) | RKey (Int -> Res e a) | RWake Int Int (Unit -> Res e a) | RPark Int (Int -> Res e a) | RAct (Co e (Res e a))
thenRes : {e : HFunctor} -> {a b : Type} -> Res e a -> (a -> Res e b) -> Res e b
alone : {e : HFunctor} -> {a : Type} -> Res e a -> Co e a
greenAlg : (e : HFunctor) -> {a : Type} -> (green <+> e).f (Res e) a -> Res e a
greenHandler : (e : HFunctor) -> Handler (green <+> e)
data Thread (e : HFunctor) a = Main (Res e a) | Other (Res e Unit)
data Step (e : HFunctor) a = SFinal a | SEnd | SYield (Unit -> Thread e a) | SSpawn (Res e Unit) (Unit -> Thread e a) | SKey (Int -> Thread e a) | SWake Int Int (Unit -> Thread e a) | SPark Int (Int -> Thread e a) | SAct (Co e (Thread e a)) | SNext
view : {e : HFunctor} -> {a : Type} -> Thread e a -> Step e a
type Queue a = { front : List a, back : List a }
emptyQ : {a : Type} -> Queue a
putQ : {a : Type} -> a -> Queue a -> Queue a
fromBack : {a : Type} -> List a -> Maybe (a, Queue a)
takeQ : {a : Type} -> Queue a -> Maybe (a, Queue a)
type Sched (e : HFunctor) a = { ready : Queue (Unit -> Thread e a), woken : OrdMap.OrdMap Int (Queue Int), parked : OrdMap.OrdMap Int (Queue (Int -> Thread e a)), next : Int, pending : Int, turns : Int }
data Joined a = Alone | Joining (Maybe a)
isEmptyQ : {a : Type} -> Queue a -> Bool
turnsBetween : Int
queueOf : {a : Type} -> Int -> OrdMap.OrdMap Int (Queue a) -> Queue a
withDone : {e : HFunctor} -> {a : Type} -> List (Thread e a) -> Sched e a -> Sched e a
ended : {e : HFunctor} -> {a : Type} -> (o : Outside e) -> o.Group (Thread e a) -> Maybe a -> Co e (Maybe a)
outcome : {e : HFunctor} -> {a : Type} -> Joined a -> Sched e a -> Maybe a
go : {e : HFunctor} -> {a : Type} -> (o : Outside e) -> o.Group (Thread e a) -> Joined a -> Sched e a -> Step e a -> Co e (Maybe a)
begun : {e : HFunctor} -> {a : Type} -> Sched e a
scheduledWith : (e : HFunctor) -> (o : Outside e) -> {a : Type} -> Co (green <+> e) a -> Co e (Maybe a)
scheduled : {e : HFunctor} -> {a : Type} -> Co (green <+> e) a -> Co e (Maybe a)
scheduledNow : {e : HFunctor} -> {a : Type} -> Co (green <+> e) a -> Co e (Maybe a)
togetherWith : (e : HFunctor) -> (o : Outside e) -> {a : Type} -> Co (green <+> e) a -> Co e (Maybe a)
together : {e : HFunctor} -> {a : Type} -> Co (green <+> e) a -> Co e (Maybe a)
