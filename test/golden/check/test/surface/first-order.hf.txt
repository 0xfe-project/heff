theory Monad (m : Type -> Type) where
  pure : {a : Type} -> a -> m a
  bind : {a b : Type} -> m a -> (a -> m b) -> m b

instance instanceMonadMaybe : Monad Maybe
instance instanceMonadList : Monad List
twice : {m : Type -> Type} -> {{mo : Monad m}} -> m Int -> m Int
r : Maybe Int
sums : List Int
none : Maybe Int
