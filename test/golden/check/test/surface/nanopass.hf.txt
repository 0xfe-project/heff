import co
import nanopass
import prelude

theory Core where
  E : Type
  P : Type
  var : String -> E
  int : Int -> E
  add : E -> E -> E
  lam : String -> E -> E
  app : E -> E -> E
  pvar : String -> P
  pint : Int -> P
  match : E -> P -> E -> E -> E

theory WithLet where
  core : Core
  letIn : String -> core.E -> core.E -> core.E

type Term = (l : Core) -> l.E
type PTerm = (l : Core) -> l.P
type LetTerm = (l : WithLet) -> l.core.E
type LetPTerm = (l : WithLet) -> l.core.P
shown : Core
shownLet : WithLet
showTerm : Term -> String
showLet : LetTerm -> String
example : LetTerm
removeLet : LetTerm -> Term
type Folded = { term : Term, asInt : Maybe Int }
type FoldedP = { pterm : PTerm, pInt : Maybe Int }
lit : Int -> Folded
opaque : Term -> Folded
folding : Core
foldConstants : Term -> Term
type Named = { lterm : LetTerm, atomic : Bool }
atom : LetTerm -> Named
compound : LetTerm -> Named
ltermOf : Named -> LetTerm
isAtomic : Named -> Bool
here : fresh <: fresh
letifying : Core
letify : Term -> LetTerm
told : stages <: stages
passes : {{st : stages <: stages}} -> LetTerm -> Co stages LetTerm
compile : LetTerm -> Co stages LetTerm
