import co
import std.handle

data GpuDevice
data GpuBuffer
data GpuPipeline
type Device = Handle GpuDevice
type Buffer = Handle GpuBuffer
type Pipeline = Handle GpuPipeline
type GpuError = { message : String }
data GpuF (m : Type -> Type) x = Open (Either GpuError Device -> x) | NewBuffer Device Int (Either GpuError Buffer -> x) | WriteBuffer Device Buffer Int Bytes (Either GpuError Unit -> x) | ReadBuffer Device Buffer Int Int (Either GpuError Bytes -> x) | Load Device String String (Either GpuError Pipeline -> x) | Dispatch Device Pipeline (List Buffer) Int Int Int (Either GpuError Unit -> x) | FreeBuffer Buffer (Unit -> x)
gpu : HFunctor
open : {e : HFunctor} -> {{ev : gpu <: e}} -> Co e (Either GpuError Device)
newBuffer : {e : HFunctor} -> {{ev : gpu <: e}} -> Device -> Int -> Co e (Either GpuError Buffer)
writeBuffer : {e : HFunctor} -> {{ev : gpu <: e}} -> Device -> Buffer -> Int -> Bytes -> Co e (Either GpuError Unit)
readBuffer : {e : HFunctor} -> {{ev : gpu <: e}} -> Device -> Buffer -> Int -> Int -> Co e (Either GpuError Bytes)
load : {e : HFunctor} -> {{ev : gpu <: e}} -> Device -> String -> String -> Co e (Either GpuError Pipeline)
dispatch : {e : HFunctor} -> {{ev : gpu <: e}} -> Device -> Pipeline -> List Buffer -> Int -> Int -> Int -> Co e (Either GpuError Unit)
freeBuffer : {e : HFunctor} -> {{ev : gpu <: e}} -> Buffer -> Co e Unit
showError : GpuError -> String
