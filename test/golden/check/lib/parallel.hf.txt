import co

data ParF a b (m : Type -> Type) x = Par (m a) (m b) (a -> b -> x)
parOf : (a : Type) -> (b : Type) -> HFunctor
parIn : {a b : Type} -> {e : HFunctor} -> {{inj : parOf a b <: e}} -> Co e a -> Co e b -> Co e (Pair a b)
inOrder : {a b x : Type} -> {e : HFunctor} -> ParF a b (Co e) x -> Co e x
inTurn : (a : Type) -> (b : Type) -> (e : HFunctor) -> Handler (parOf a b <+> e)
runPar : {a b x : Type} -> ParF a b (\y -> y) x -> x
turns : (a : Type) -> (b : Type) -> Handler (parOf a b)
bothSides : (Int -> Co (parOf Int Int <+> parOf Int Int) Int) -> Co (parOf Int Int <+> parOf Int Int) (Pair Int Int)
