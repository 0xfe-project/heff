import co
import std.handle

data Element
type Node = Handle Element
data DomF (m : Type -> Type) x = Create String (Node -> x) | Text Node String x | Attr Node String String x | Append Node Node x | Clear Node x | Value Node (String -> x) | On Node String (String -> m Unit) x | Select String (Maybe Node -> x) | XPath String (Maybe Node -> x) | Sleep Int x | Fetch String (String -> x)
dom : HFunctor
create : {e : HFunctor} -> {{ev : dom <: e}} -> String -> Co e Node
text : {e : HFunctor} -> {{ev : dom <: e}} -> Node -> String -> Co e Unit
attr : {e : HFunctor} -> {{ev : dom <: e}} -> Node -> String -> String -> Co e Unit
append : {e : HFunctor} -> {{ev : dom <: e}} -> Node -> Node -> Co e Unit
clear : {e : HFunctor} -> {{ev : dom <: e}} -> Node -> Co e Unit
value : {e : HFunctor} -> {{ev : dom <: e}} -> Node -> Co e String
on : {e : HFunctor} -> {{ev : dom <: e}} -> Node -> String -> (String -> Co e Unit) -> Co e Unit
select : {e : HFunctor} -> {{ev : dom <: e}} -> String -> Co e (Maybe Node)
xpath : {e : HFunctor} -> {{ev : dom <: e}} -> String -> Co e (Maybe Node)
found : {e : HFunctor} -> Maybe Node -> (Node -> Co e Unit) -> Co e Unit
sleep : {e : HFunctor} -> {{ev : dom <: e}} -> Int -> Co e Unit
fetchText : {e : HFunctor} -> {{ev : dom <: e}} -> String -> Co e String
textIn : {e : HFunctor} -> {{ev : dom <: e}} -> Node -> String -> String -> Co e Node
