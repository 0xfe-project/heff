theory HFunctor where
  f : (Type -> Type) -> Type -> Type
  hfmap : {m : Type -> Type} -> ({a b : Type} -> (a -> b) -> m a -> m b) -> {a b : Type} -> (a -> b) -> f m a -> f m b
  hmap : {m : Type -> Type} -> {n : Type -> Type} -> ({x : Type} -> m x -> n x) -> {a : Type} -> f m a -> f n a

theory Handler (h : HFunctor) where
  M : Type -> Type
  ret : {a : Type} -> a -> M a
  bind : {a b : Type} -> M a -> (a -> M b) -> M b
  malg : {a : Type} -> h.f M a -> M a

type Co (h : HFunctor) a = (hd : Handler h) -> (r : Type) -> (a -> hd.M r) -> hd.M r
val : {h : HFunctor} -> {a : Type} -> a -> Co h a
bind : {h : HFunctor} -> {a b : Type} -> Co h a -> (a -> Co h b) -> Co h b
pure : {h : HFunctor} -> {a : Type} -> a -> Co h a
handle : {h : HFunctor} -> {a : Type} -> (hd : Handler h) -> Co h a -> hd.M a
op : {h : HFunctor} -> {a b : Type} -> h.f (Co h) a -> (a -> Co h b) -> Co h b
data SumF (f : (Type -> Type) -> Type -> Type) (g : (Type -> Type) -> Type -> Type) (m : Type -> Type) a = InL (f m a) | InR (g m a)
plus : HFunctor -> HFunctor -> HFunctor
(<+>) : HFunctor -> HFunctor -> HFunctor
(<:) : HFunctor -> HFunctor -> Type
here : (h : HFunctor) -> h <: h
inl : (h : HFunctor) -> (k : HFunctor) -> h <: plus h k
inr : (h : HFunctor) -> (k : HFunctor) -> k <: plus h k
lift : {h : HFunctor} -> {k : HFunctor} -> {a : Type} -> (h <: k) -> Co h a -> Co k a
through : (h : HFunctor) -> (e : HFunctor) -> ({a : Type} -> h.f (Co e) a -> Co e a) -> Handler (plus h e)
extend : (k : HFunctor) -> (h : HFunctor) -> (hd : Handler h) -> ({a : Type} -> k.f hd.M a -> hd.M a) -> Handler (plus k h)
