import co
import prelude

data ExnF (m : Type -> Type) x = Throw String | Catch (m x) (String -> m x)
exn : HFunctor
throw : {e : HFunctor} -> {a : Type} -> {{ev : exn <: e}} -> String -> Co e a
catch : {e : HFunctor} -> {a : Type} -> {{ev : exn <: e}} -> Co e a -> (String -> Co e a) -> Co e a
