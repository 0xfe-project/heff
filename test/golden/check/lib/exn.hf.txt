import co

data ExnF err (m : Type -> Type) x = Throw err | Catch (m x) (err -> m x)
exn : (err : Type) -> HFunctor
throw : {err : Type} -> {e : HFunctor} -> {a : Type} -> {{ev : exn err <: e}} -> err -> Co e a
catch : {err : Type} -> {e : HFunctor} -> {a : Type} -> {{ev : exn err <: e}} -> Co e a -> (err -> Co e a) -> Co e a
data DivByZero = MkDivByZero
exnDiv : {a : Type} -> {e : HFunctor} -> {{div : Division a}} -> {{z : IsZero a}} -> {{ev : exn DivByZero <: e}} -> a -> a -> Co e a
exnMod : {a : Type} -> {e : HFunctor} -> {{mod : Remainder a}} -> {{z : IsZero a}} -> {{ev : exn DivByZero <: e}} -> a -> a -> Co e a
triedFunctor : (err : Type) -> (e : HFunctor) -> RawFunctor
attempted : (err : Type) -> (e : HFunctor) -> Handler (plus (exn err) e)
attempt : (err : Type) -> (e : HFunctor) -> {a : Type} -> Co (plus (exn err) e) a -> Co e (Either err a)
