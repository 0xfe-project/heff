import co
import green

data Group t = MkGroup Int
type Begun (m : Type -> Type) x = (r : Type) -> ((t : Type) -> m t -> (Maybe t -> x) -> r) -> r
type Finished x = (r : Type) -> ((t : Type) -> (List t -> x) -> r) -> r
begunWith : {m : Type -> Type} -> {t x : Type} -> m t -> (Maybe t -> x) -> Begun m x
finishedWith : {t x : Type} -> (List t -> x) -> Finished x
data ReactorF (m : Type -> Type) x = Open (Int -> x) | Start Int (Begun m x) | Finish Int Bool (Finished x) | Close Int x
reactor : HFunctor
open : {e : HFunctor} -> {{rx : reactor <: e}} -> {t : Type} -> Co e (Group t)
start : {e : HFunctor} -> {{rx : reactor <: e}} -> {t : Type} -> Group t -> Co e t -> Co e (Maybe t)
finished : {e : HFunctor} -> {{rx : reactor <: e}} -> {t : Type} -> Group t -> Bool -> Co e (List t)
close : {e : HFunctor} -> {{rx : reactor <: e}} -> {t : Type} -> Group t -> Co e Unit
outside : {e : HFunctor} -> {{rx : reactor <: e}} -> Outside e
scheduled : (e : HFunctor) -> {{rx : reactor <: e}} -> {a : Type} -> Co (green <+> e) a -> Co e (Maybe a)
