import co

data FsF (m : Type -> Type) x = ReadFile String (Maybe String -> x) | WriteFile String String (Bool -> x) | MkDir String (Bool -> x) | ReadDir String (Maybe (List String) -> x) | Remove String (Bool -> x) | ReadOnly String (Bool -> x) | Copy String String (Bool -> x) | Move String String (Bool -> x)
fs : HFunctor
readFile : {e : HFunctor} -> {{ev : fs <: e}} -> String -> Co e (Maybe String)
writeFile : {e : HFunctor} -> {{ev : fs <: e}} -> String -> String -> Co e Bool
mkDir : {e : HFunctor} -> {{ev : fs <: e}} -> String -> Co e Bool
readDir : {e : HFunctor} -> {{ev : fs <: e}} -> String -> Co e (Maybe (List String))
remove : {e : HFunctor} -> {{ev : fs <: e}} -> String -> Co e Bool
readOnly : {e : HFunctor} -> {{ev : fs <: e}} -> String -> Co e Bool
copy : {e : HFunctor} -> {{ev : fs <: e}} -> String -> String -> Co e Bool
move : {e : HFunctor} -> {{ev : fs <: e}} -> String -> String -> Co e Bool
