import co
import std.handle

data Ended = Exited Int | Signalled Int | TimedOut | NotRun
data Input = NoInput | Given String | Inherited
data Output = Collected | Inherited
type Invocation = { command : String, args : List String = [], dir : String = "", env : List (Pair String String) = [], input : Input = NoInput, output : Output = Collected, errors : Output = Collected, timeout : Int = 0, keep : Int = 0 }
type Ran = { ended : Ended, out : String, err : String }
endedWell : Ended -> Bool
succeeded : Ran -> Bool
data Started
type Subprocess = Handle Started
data ProcessF (m : Type -> Type) x = Args (List String -> x) | GetEnv String (Maybe String -> x) | ProgramPath (String -> x) | WorkingDir (String -> x) | Platform (String -> x) | Exit Int x | Exec String (List String) x | Run Invocation (Ran -> x) | Start Invocation (Maybe Subprocess -> x) | Wait Subprocess (Ran -> x) | WaitAny (List Subprocess) (Maybe (Pair Subprocess Ran) -> x) | TryWait Subprocess (Maybe Ran -> x) | Kill Subprocess x | Terminate Subprocess (Ran -> x)
process : HFunctor
args : {e : HFunctor} -> {{ev : process <: e}} -> Co e (List String)
getEnv : {e : HFunctor} -> {{ev : process <: e}} -> String -> Co e (Maybe String)
programPath : {e : HFunctor} -> {{ev : process <: e}} -> Co e String
workingDir : {e : HFunctor} -> {{ev : process <: e}} -> Co e String
platform : {e : HFunctor} -> {{ev : process <: e}} -> Co e String
exit : {e : HFunctor} -> {{ev : process <: e}} -> Int -> Co e Unit
exec : {e : HFunctor} -> {{ev : process <: e}} -> String -> List String -> Co e Unit
run : {e : HFunctor} -> {{ev : process <: e}} -> Invocation -> Co e Ran
start : {e : HFunctor} -> {{ev : process <: e}} -> Invocation -> Co e (Maybe Subprocess)
wait : {e : HFunctor} -> {{ev : process <: e}} -> Subprocess -> Co e Ran
waitAny : {e : HFunctor} -> {{ev : process <: e}} -> List Subprocess -> Co e (Maybe (Pair Subprocess Ran))
tryWait : {e : HFunctor} -> {{ev : process <: e}} -> Subprocess -> Co e (Maybe Ran)
kill : {e : HFunctor} -> {{ev : process <: e}} -> Subprocess -> Co e Unit
terminate : {e : HFunctor} -> {{ev : process <: e}} -> Subprocess -> Co e Ran
