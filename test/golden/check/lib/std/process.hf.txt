import co
import std.handle

data Ended = Exited Int | Signalled Int | TimedOut | NotRun
type Invocation = { command : String, args : List String = [], dir : String = "", env : List (Pair String String) = [], input : Maybe String = Nothing, timeout : Int = 0, keep : Int = 0 }
type Ran = { ended : Ended, out : String, err : String }
endedWell : Ended -> Bool
succeeded : Ran -> Bool
data Started
type Child = Handle Started
data ProcessF (m : Type -> Type) x = Args (List String -> x) | GetEnv String (Maybe String -> x) | ProgramPath (String -> x) | WorkingDir (String -> x) | Platform (String -> x) | Exit Int x | Exec String (List String) x | Run Invocation (Ran -> x) | Launch String (List String) (Maybe Child -> x) | Poll Child (Maybe Int -> x) | Terminate Child (Int -> x)
process : HFunctor
args : {e : HFunctor} -> {{ev : process <: e}} -> Co e (List String)
getEnv : {e : HFunctor} -> {{ev : process <: e}} -> String -> Co e (Maybe String)
programPath : {e : HFunctor} -> {{ev : process <: e}} -> Co e String
workingDir : {e : HFunctor} -> {{ev : process <: e}} -> Co e String
platform : {e : HFunctor} -> {{ev : process <: e}} -> Co e String
exit : {e : HFunctor} -> {{ev : process <: e}} -> Int -> Co e Unit
exec : {e : HFunctor} -> {{ev : process <: e}} -> String -> List String -> Co e Unit
run : {e : HFunctor} -> {{ev : process <: e}} -> Invocation -> Co e Ran
launch : {e : HFunctor} -> {{ev : process <: e}} -> String -> List String -> Co e (Maybe Child)
exited : {e : HFunctor} -> {{ev : process <: e}} -> Child -> Co e (Maybe Int)
terminate : {e : HFunctor} -> {{ev : process <: e}} -> Child -> Co e Int
