import co

data Reference s = MkReference Int
data ReferenceF s (m : Type -> Type) x = NewReference s (Reference s -> x) | ReadReference (Reference s) (s -> x) | WriteReference (Reference s) s x
reference : (s : Type) -> HFunctor
newReference : {s : Type} -> {e : HFunctor} -> {{ev : reference s <: e}} -> s -> Co e (Reference s)
readReference : {s : Type} -> {e : HFunctor} -> {{ev : reference s <: e}} -> Reference s -> Co e s
writeReference : {s : Type} -> {e : HFunctor} -> {{ev : reference s <: e}} -> Reference s -> s -> Co e Unit
