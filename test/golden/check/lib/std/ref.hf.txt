import co

data Ref s = MkRef Int
data RefF s (m : Type -> Type) x = NewRef s (Ref s -> x) | ReadRef (Ref s) (s -> x) | WriteRef (Ref s) s x
ref : (s : Type) -> HFunctor
newRef : {s : Type} -> {e : HFunctor} -> {{ev : ref s <: e}} -> s -> Co e (Ref s)
readRef : {s : Type} -> {e : HFunctor} -> {{ev : ref s <: e}} -> Ref s -> Co e s
writeRef : {s : Type} -> {e : HFunctor} -> {{ev : ref s <: e}} -> Ref s -> s -> Co e Unit
